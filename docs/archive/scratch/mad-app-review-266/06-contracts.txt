import { Injectable, Logger, NotFoundException } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { TenantPrismaService } from '../common/tenant/tenant-prisma.service';
import { FilesService } from '../files/files.service';
import { MSG } from '../common/i18n/messages';

const ATTACHMENT_SELECT = {
  id: true,
  claimId: true,
  filename: true,
  size: true,
  mimeType: true,
  uploadedAt: true,
} as const;

/**
 * مرفقات المطالبة (§16 + احتياجات §11): فاتورة الإصلاح، صور الضرر، المراسلات.
 *
 * Mirror of ContractAttachmentsService — bytes in object storage, tenant-scoped
 * metadata here, every method inside the claim's own tenant context so RLS
 * makes a neighbour's claim (and its files) unreachable even by guessed id.
 */
@Injectable()
export class ClaimAttachmentsService {
  private readonly logger = new Logger(ClaimAttachmentsService.name);

  constructor(
    private readonly tenantPrisma: TenantPrismaService,
    private readonly files: FilesService,
  ) {}

  list(companyId: string, claimId: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const claim = await tx.contractClaim.findUnique({ where: { id: claimId }, select: { id: true } });
      if (!claim) throw new NotFoundException(MSG.claim_not_found);
      return tx.claimAttachment.findMany({ where: { claimId }, select: ATTACHMENT_SELECT, orderBy: { uploadedAt: 'desc' } });
    });
  }

  async add(companyId: string, claimId: string, file: { originalname: string; mimetype: string; buffer: Buffer; size: number }) {
    // Upload first: allow-list, size cap and key-namespacing live in
    // FilesService, so a rejected file never reaches the database.
    const stored = await this.files.upload(companyId, file);
    try {
      return await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const claim = await tx.contractClaim.findUnique({ where: { id: claimId }, select: { id: true } });
        if (!claim) throw new NotFoundException(MSG.claim_not_found);
        return tx.claimAttachment.create({
          // companyId omitted on purpose — injected by the tenant-scope extension.
          data: { claimId, fileKey: stored.key, filename: stored.filename, size: stored.size, mimeType: stored.mimeType } as unknown as Prisma.ClaimAttachmentUncheckedCreateInput,
          select: ATTACHMENT_SELECT,
        });
      });
    } catch (e) {
      // Row not written — don't leave the object orphaned in storage.
      await this.files.remove(companyId, stored.key).catch((err: Error) => this.logger.warn(`Could not roll back orphaned upload "${stored.key}": ${err.message}`));
      throw e;
    }
  }

  downloadUrl(companyId: string, attachmentId: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const att = await tx.claimAttachment.findUnique({ where: { id: attachmentId }, select: { fileKey: true } });
      if (!att) throw new NotFoundException(MSG.file_not_found);
      return this.files.downloadUrl(companyId, att.fileKey);
    });
  }

  async remove(companyId: string, attachmentId: string): Promise<void> {
    const fileKey = await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const att = await tx.claimAttachment.findUnique({ where: { id: attachmentId }, select: { fileKey: true } });
      if (!att) throw new NotFoundException(MSG.file_not_found);
      await tx.claimAttachment.delete({ where: { id: attachmentId } });
      return att.fileKey;
    });
    await this.files.remove(companyId, fileKey).catch((e: Error) => this.logger.warn(`Could not delete attachment object "${fileKey}": ${e.message}`));
  }
}

import { ForbiddenException, Injectable, Logger, NotFoundException } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { TenantPrismaService } from '../common/tenant/tenant-prisma.service';
import { FilesService, isSensitiveKey } from '../files/files.service';
import { MSG } from '../common/i18n/messages';

/** What the API returns for one attachment — never the raw storage key alone,
 * since a key is only useful through a signed download URL fetched on demand. */
const ATTACHMENT_SELECT = {
  id: true,
  contractId: true,
  filename: true,
  size: true,
  mimeType: true,
  uploadedAt: true,
} as const;

/**
 * Files attached to a lease — the signed PDF, an annex, a clearance letter.
 * Deliberately not "an ID scan": see the ContractAttachment model comment.
 *
 * The bytes live in object storage; this keeps the tenant-scoped metadata row
 * and is the single place that authorizes access. Every method runs inside the
 * contract's own tenant context, so RLS guarantees one company can never list,
 * download, or delete another's attachment even by guessing an id.
 */
@Injectable()
export class ContractAttachmentsService {
  private readonly logger = new Logger(ContractAttachmentsService.name);

  constructor(
    private readonly tenantPrisma: TenantPrismaService,
    private readonly files: FilesService,
  ) {}

  list(companyId: string, contractId: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      // A contract that isn't this company's is invisible under RLS, so this
      // both proves it exists and that the caller owns it.
      const contract = await tx.contract.findUnique({ where: { id: contractId }, select: { id: true } });
      if (!contract) throw new NotFoundException(MSG.contract_not_found);
      return tx.contractAttachment.findMany({ where: { contractId }, select: ATTACHMENT_SELECT, orderBy: { uploadedAt: 'desc' } });
    });
  }

  async add(companyId: string, contractId: string, file: { originalname: string; mimetype: string; buffer: Buffer; size: number }) {
    // Upload first: the MIME allow-list, size cap and key-namespacing all live in
    // FilesService, so a rejected file never reaches the database.
    // ⚖️ ق-١٣٦٥ — **إلى المخزن الحسّاس** («مرفقاتُ العقود حسّاسةٌ الآن … والحساسيةُ بالمخزن لا بسؤال الرافع»): يُغلَّف قبل أن يغادر
    // الخادم، ولا يُسأل الرافعُ أحسّاسٌ هو — العقدُ يحمل هويّةَ المستأجر وعنوانَه فكلُّ مرفقاته في المخزن نفسِه.
    const stored = await this.files.uploadSensitive(companyId, file);
    try {
      return await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const contract = await tx.contract.findUnique({ where: { id: contractId }, select: { id: true } });
        if (!contract) throw new NotFoundException(MSG.contract_not_found);
        return tx.contractAttachment.create({
          // companyId omitted on purpose — injected per-row by the tenant scope
          // extension (see accounts.service.ts for why this cast exists). Every
          // other required column is passed explicitly, so the cast hides
          // nothing but the injected tenant id.
          data: { contractId, fileKey: stored.key, filename: stored.filename, size: stored.size, mimeType: stored.mimeType } as unknown as Prisma.ContractAttachmentUncheckedCreateInput,
          select: ATTACHMENT_SELECT,
        });
      });
    } catch (e) {
      // The row didn't get written (bad contract id, or a race). Don't leave the
      // uploaded object orphaned in storage.
      await this.files.remove(companyId, stored.key).catch((err: Error) => this.logger.warn(`Could not roll back orphaned upload "${stored.key}": ${err.message}`));
      throw e;
    }
  }

  /**
   * ⚖️ ق-١٣٦٥ — **لا رابطَ موقَّعاً لمرفق عقد** («والتنزيلُ من المخزن الحسّاس عبر الخادم يفكّ ويبثّ، لا برابطٍ موقَّع»). والرفضُ
   * للفئة كلِّها لا للمغلَّف وحدَه: ما سبق الغلافَ مرفقُ عقدٍ كذلك، ويُبثّ عبر الخادم مثلَه (`download`). والعزلُ قبل الرفض —
   * مرفقُ منشأةٍ أخرى `404` كما كان، فلا يُعرف وجودُه.
   */
  downloadUrl(companyId: string, attachmentId: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const att = await tx.contractAttachment.findUnique({ where: { id: attachmentId }, select: { fileKey: true } });
      if (!att) throw new NotFoundException(MSG.file_not_found);
      throw new ForbiddenException(MSG.file_sensitive_server_only);
    });
  }

  /**
   * ⚖️ ق-١٣٦٥ — **التنزيلُ عبر الخادم**: يُقرأ الصفُّ تحت سياق المنشأة (فمرفقُ غيرها غيرُ موجود)، ثم يُفكّ الغلافُ ويُعاد البايتاتُ
   * بنوعها واسمها ليبثّها المتحكّم. وما سبق الغلافَ (مخزنٌ عاديّ) يُقرأ كما هو — يُعدّ في الهجرة ولا يُحوَّل.
   */
  async download(companyId: string, attachmentId: string): Promise<{ buffer: Buffer; filename: string; mimeType: string }> {
    const att = await this.tenantPrisma.runInTenantContext(companyId, (tx) =>
      tx.contractAttachment.findUnique({ where: { id: attachmentId }, select: { fileKey: true, filename: true, mimeType: true } }),
    );
    if (!att) throw new NotFoundException(MSG.file_not_found);
    const buffer = isSensitiveKey(att.fileKey) ? await this.files.readSensitive(companyId, att.fileKey) : await this.files.readPlain(companyId, att.fileKey);
    return { buffer, filename: att.filename, mimeType: att.mimeType };
  }

  async remove(companyId: string, attachmentId: string): Promise<void> {
    const fileKey = await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const att = await tx.contractAttachment.findUnique({ where: { id: attachmentId }, select: { fileKey: true } });
      if (!att) throw new NotFoundException(MSG.file_not_found);
      await tx.contractAttachment.delete({ where: { id: attachmentId } });
      return att.fileKey;
    });
    // Row is gone; clear the object too. A failure here only orphans bytes, so
    // it's logged rather than surfaced as an error to the user.
    await this.files.remove(companyId, fileKey).catch((e: Error) => this.logger.warn(`Could not delete attachment object "${fileKey}": ${e.message}`));
  }
}

  @RequiresAction('view')
  attachmentDownloadUrl(@CurrentTenant() tenant: TenantContext, @Param('attachmentId') attachmentId: string) {
    return this.attachments.downloadUrl(tenant.companyId, attachmentId);
  }

  /**
   * ⚖️ ق-١٣٦٥ — **التنزيلُ عبر الخادم يفكّ ويبثّ** (لا رابطَ موقَّعاً لمرفق عقد). بنوعه واسمه، ولا يُخزَّن في وسيطٍ ولا متصفّح
   * (`no-store`) — فالمفكوكُ لا يبقى في نسخةٍ خارج المخزن.
   */
  @Get('attachments/:attachmentId/download')
  @RequiresAction('view')
  async downloadAttachment(@CurrentTenant() tenant: TenantContext, @Param('attachmentId') attachmentId: string, @Res({ passthrough: true }) res: Response) {
    const file = await this.attachments.download(tenant.companyId, attachmentId);
    res.setHeader('Cache-Control', 'no-store');
    return new StreamableFile(file.buffer, {
      type: file.mimeType,
      length: file.buffer.length,
      disposition: `attachment; filename="file"; filename*=UTF-8''${encodeURIComponent(file.filename)}`,
    });
  }

  @Delete('attachments/:attachmentId')
  @RequiresAction('delete')
  @HttpCode(204)

/**
 * **العميلُ مصدرُ الحقيقة لاسم المستأجر وهاتفه** (`ق-١٠٢٨`/①⑥): الحقلان على العقد ذاكرةٌ مشتقّةٌ تُكتب
 * عند الربط وتُحدَّث في معاملة تعديل العميل — وعند القراءة يُقدَّم ما في بطاقته، فيُصحَّح ما افترق
 * قبل هذا البناء بالاشتقاق لا بهجرةٍ تكتبه. عقدٌ بلا عميل (مسوّدةٌ قديمة) يُقرأ كما كُتب.
 */
function withPartyFromCustomer<T extends { tenant: string; phone: string | null; customer?: { name: string; phone: string | null } | null }>(contract: T): T {
  if (!contract.customer) return contract;
  return { ...contract, tenant: contract.customer.name, phone: contract.customer.phone };
}

/** Unpaid rent settles a receivable that was already recognized as revenue;
 * everything else is income the landlord keeps. This split is the whole reason
 * a deduction carries a kind rather than just an amount. */
function deductionTarget(kind: DeductionKind): 'receivable' | 'income' {
  return kind === 'UNPAID_RENT' ? 'receivable' : 'income';
}

// ⛔ كانت هنا خريطةُ كلماتٍ خمس تحوّل `Unit.type` الحرَّ إلى تعداد العقار لفحص اتساقٍ عند
// إنشاء العقد — **وتتخطّى ما لا تعرفه صامتةً** (`office` بالحرف اللاتينيّ غيرُ مرئيٍّ لها).
// حُذفت بحكم المالك (`ق-٨٩٨`): التصنيفُ حقلٌ مهيكلٌ إلزاميٌّ على العقد (`Contract.classification`)
// **ولا يُستنبَط من نصّ** — ولا تُبقى الخريطةُ «لفحص الاتساق» فبقاؤها يُبقي الاستنباط.
// حارسٌ يُسقط الحزمة على عودتها: `no-word-classification.spec.ts`.

@Injectable()
export class ContractsService {
  private readonly logger = new Logger(ContractsService.name);

  constructor(
    private readonly tenantPrisma: TenantPrismaService,
    private readonly files: FilesService,
  ) {}

  list(companyId: string, query: ListContractsQueryDto = {}) {
    return this.tenantPrisma.runInTenantContext(companyId, (tx) => this.listRefreshed(tx, query));
  }

  findOne(companyId: string, id: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      await this.refreshInstallmentStatuses(tx, id);
      const contract = await tx.contract.findUnique({ where: { id }, include: CONTRACT_INCLUDE });
      if (!contract) throw new NotFoundException(MSG.contract_not_found);
      // العميلُ مصدرُ الحقيقة (`ق-١٠٢٨`/⑥): ما افترق على العقد قبل هذا البناء يُصحَّح بالاشتقاق عند
      // القراءة — لا بهجرةٍ تكتبه — وما بعده تُبقيه الكتابةُ متطابقاً.
      return this.withComputedFigures(withPartyFromCustomer(contract));
    });
  }

  /**
   * **الأرقامُ التي كانت تجمعها نافذةُ العقد بنفسها** (`ق-١٢٤٣`: الشاشةُ تقرأ ولا تحسب) — نصّاً بخانتين:
   * لكل قسطٍ **إجماليُّه** بالضريبة من الدالّة نفسِها التي يقيس بها الاستحقاقُ والقبض (`rentAmountsOf`؛
   * والقديمُ غيرُ المصنَّف بمبلغه — **وعقارُ التشغيل بلا ضريبة**، وهو ما كانت الشاشةُ تخطئه) **ومسدَّدُه**
   * (صافي دفعاته + خصمُها — كما تُشتقّ حالتُه) **ومتبقّيه** (موقَّعاً: الزائدُ سالب)؛ ومجموعُ المطالبات
   * المفتوحة؛ والتأميناتُ المحجوزة لدى المنشأة.
   */
  private withComputedFigures<
    T extends Pick<Contract, 'classification' | 'vatInclusion'> & {
      unit: { property: { ownership: string } };
      installments: { id: string; amount: Prisma.Decimal }[];
      payments: { installmentId: string | null; netAmount: Prisma.Decimal; discount: Prisma.Decimal }[];
      claims: { status: string; amount: Prisma.Decimal }[];
      deposits: { status: string; holder: string; amount: Prisma.Decimal }[];
    },
  >(contract: T) {
    // المعاملةُ الضريبية من حقلَي العقد المُعلَنَين (`ق-٨٩١`) — لا استنباطَ من كلمة.
    const treatment = rentTaxTreatment(contract.classification as RentClassification, contract.vatInclusion as RentVatInclusion, contract.unit.property.ownership);
    const settledBy = new Map<string, number>();
    for (const p of contract.payments) {
      if (!p.installmentId) continue;
      settledBy.set(p.installmentId, (settledBy.get(p.installmentId) ?? 0) + Number(p.netAmount) + Number(p.discount));
    }
    return {
      ...contract,
      installments: contract.installments.map((i) => {
        const gross = treatment ? rentAmountsOf(Number(i.amount), treatment).gross : round2(Number(i.amount));
        const settled = round2(settledBy.get(i.id) ?? 0);
        return { ...i, gross: money(gross), settled: money(settled), remaining: money(gross - settled) };
      }),
      claimsOutstanding: money(contract.claims.filter((c) => c.status === 'OPEN').reduce((s, c) => s + Number(c.amount), 0)),
      depositsHeld: money(contract.deposits.filter((d) => d.status === 'HELD' && d.holder === 'COMPANY').reduce((s, d) => s + Number(d.amount), 0)),
    };
  }

  create(companyId: string, dto: CreateContractDto) {
    // Retries the whole transaction on a "no" collision from a concurrent
    // create — same reasoning as InvoicesService.create.
    // ⚠️ مهلةٌ أطول من الافتراض: عقدٌ يُدخَل بتاريخٍ مضى يستحقّ أقساطَه التي حلّت في المعاملة
    // نفسِها (ق-٩٢٧/٥) — وقيدٌ لكلِّ قسط؛ فمهلةُ الخمس ثوانٍ تُسقط ترحيلَ عقدٍ بعشرين قسطاً.
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(
        companyId,
        async (tx) => {
        // بلا عملةٍ صريحة ⟵ عملةُ أساس المنشأة لا الريال (`ق-١٣٦٨`).
        const { currency, exchangeRate } = await resolveCurrencyWithin(tx, companyId, dto);
        const unit = await tx.unit.findUnique({ where: { id: dto.unitId }, include: { property: true } });
        if (!unit) throw new NotFoundException(MSG.unit_not_found);

        const start = new Date(dto.start);
        const end = new Date(dto.end);
        const isDraft = dto.draft ?? false;
        // **العميلُ يُختار بمعرّفه — للمسوّدة كما للحيّ** (`ق-١٠٢٨`/①②): لا بحثَ بالاسم ولا إنشاءَ ضمنيّ.
        const customer = await this.linkableCustomer(tx, dto.customerId);
        if (!isDraft) await this.validateContract(tx, unit, unit.property, start, end, customer.name);

        const no = await this.nextContractNumber(tx, companyId, start);

        const created = await tx.contract.create({
          data: {
            unitId: dto.unitId,
            customerId: customer.id,
            no,
            ejarNo: dto.ejarNo,
            // مشتقّان من بطاقة العميل — ذاكرةٌ تُحدَّث مع تعديله، لا لقطةٌ تشيخ (`ق-١٠٢٨`/①⑥).
            tenant: customer.name,
            phone: customer.phone,
            value: dto.value,
            cycle: dto.cycle,
            start,
            end,
            deposit: dto.deposit ?? 0,
            services: dto.services,
            furnished: dto.furnished,
            // إلزاميان من الـDTO وحدَه — لا افتراضَ ولا استنباط (ق-٨٩١ · ق-٨٩٢).
            classification: dto.classification,
            vatInclusion: dto.vatInclusion,
            status: isDraft ? 'DRAFT' : 'DUE',
            currency,
            exchangeRate,
            baseValue: toBaseAmount(dto.value, exchangeRate),
          } as unknown as Prisma.ContractUncheckedCreateInput,
        });

        if (!isDraft) {
          const schedule = generateInstallmentSchedule(start, end, dto.value, dto.cycle);
          await tx.installment.createMany({
            // Installments inherit the contract's currency and rate — they're
            // just a schedule of the same obligation, not separate deals.
            data: schedule.map((s) => ({
              contractId: created.id,
              dueDate: s.dueDate,
              amount: s.amount,
              // فترةُ الخدمة تُثبَّت لحظةَ التوليد ولا تُشتقّ لاحقاً (ق-٤٧٥/②).
              serviceStart: s.serviceStart,
              serviceEnd: s.serviceEnd,
              currency,
              exchangeRate,
              baseAmount: toBaseAmount(s.amount, exchangeRate),
            })) as unknown as Prisma.InstallmentUncheckedCreateInput[],
          });

          // Matches the mockup: creating a live (non-draft) contract for a
          // unit that carries a matching-name active reservation converts
          // that reservation rather than leaving it dangling — mismatched
          // names were already rejected above by validateContract.
          const rsv = await activeReservationForUnit(tx, dto.unitId);
          if (rsv) await tx.reservation.update({ where: { id: rsv.id }, data: { status: 'CONVERTED' } });

          // ⭐ ما حلّ استحقاقُه من الجدول يُستحقّ ذمّةً في اللحظة نفسِها (ق-٩١٧ · ق-٩٢٧/٥):
          // عقدٌ يُدخَل بتاريخٍ مضى يحمل أقساطاً حلّت، وكلٌّ منها نقطةُ ضريبةٍ بتاريخها —
          // لا ينتظر الكنسَ اليوميّ. والمقفلُ من الفترات يُتخطّى ويُعدّ لا يوقف الإنشاء.
          await accrueDueWithinTx(tx, companyId, new Date(), created.id);
        }

        return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: created.id }, include: CONTRACT_INCLUDE }));
        },
        ACCRUAL_TX_OPTIONS,
      ),
    );
  }

  update(companyId: string, id: string, dto: UpdateContractDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await tx.contract.findUnique({ where: { id } });
      if (!existing) throw new NotFoundException(MSG.contract_not_found);

      // عقدٌ سبق الهجرة يحمل قيمةً قديمةً معلَنة — وأوّلُ تعديلٍ له يوجب إعلانَ ما لم
      // يُعلَن (ق-٨٩٨): «يُعلَن ولا يُصحَّح رجعياً، ويُنقص بيدٍ عند لمس العقد».
      if (existing.classification === 'LEGACY_UNCLASSIFIED' && dto.classification === undefined) {
        throw new BadRequestException(MSG.contract_classification_required);
      }
      if (existing.vatInclusion === 'LEGACY_UNDECLARED' && dto.vatInclusion === undefined) {
        throw new BadRequestException(MSG.contract_vat_inclusion_required);
      }

      const data: Prisma.ContractUncheckedUpdateInput = {};
      if (dto.classification !== undefined) data.classification = dto.classification;
      if (dto.vatInclusion !== undefined) data.vatInclusion = dto.vatInclusion;
      // **تغييرُ الطرف قبل أوّل ترحيلٍ فقط** (`ق-١٠٢٨`/③): بعده ذمّةٌ وقيودٌ نُسبت إليه — استحقاقٌ رُحِّل،
      // أو قبضٌ، أو وديعةٌ أو مطالبةٌ أو خلوّ — وتحويلُها إعادةُ كتابة تاريخ (`§6.4`). والمخرجُ تسمّيه
      // الرسالة: إنهاءُ العقد وعقدٌ جديد. أمّا الاسمُ والهاتفُ فمشتقّان من العميل لا مُدخَلان.
      if (dto.customerId !== undefined && dto.customerId !== existing.customerId) {
        const party = await this.linkableCustomer(tx, dto.customerId);
        const [accrued, paid, deposits, claims, keyMoney] = await Promise.all([
          tx.rentAccrual.count({ where: { contractId: id, journalEntryId: { not: null } } }),
          tx.rentPayment.count({ where: { contractId: id } }),
          tx.contractDeposit.count({ where: { contractId: id } }),
          tx.contractClaim.count({ where: { contractId: id } }),
          tx.contractKeyMoney.count({ where: { contractId: id } }),
        ]);
        if (accrued + paid + deposits + claims + keyMoney > 0) throw new ConflictException(MSG.contract_party_locked_after_posting);
        data.customerId = party.id;
        data.tenant = party.name;
        data.phone = party.phone;
      }
      if (dto.services !== undefined) data.services = dto.services;
      if (dto.furnished !== undefined) data.furnished = dto.furnished;
      if (dto.ejarNo !== undefined) data.ejarNo = dto.ejarNo;

      return this.withComputedFigures(await tx.contract.update({ where: { id }, data, include: CONTRACT_INCLUDE }));
    });
  }

  cancel(companyId: string, id: string, dto: CancelContractDto) {
    // Retries on a journal-entry "no" collision when a settlement is posted.
    // ⚠️ ومهلةُ الدفعة كالإنشاء: الإلغاءُ يعترف بما قُدِّم حتى تاريخه ويعكس المُسقَط — قيدٌ لكلِّ
    // شهرٍ ولكلِّ قسطٍ مُسقَط، وعقدٌ طويل يتجاوز بها الخمسَ ثوانٍ.
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(
        companyId,
        async (tx) => {
        const existing = await tx.contract.findUnique({ where: { id } });
        if (!existing) throw new NotFoundException(MSG.contract_not_found);
        if (existing.status === 'CANCELLED') throw new ConflictException(MSG.contract_already_cancelled);

        const cancelDate = new Date(dto.cancelDate);
        const accrualContract = await tx.contract.findUniqueOrThrow({ where: { id }, select: ACCRUAL_CONTRACT_SELECT });

        // Worked out **before** the installments are touched: 'all' and
        // 'installments' read the very rows that `installmentDisposition:
        // 'cancel'` is about to void, and voiding them first would settle at
        // zero.
        // ⚖️ **قبل كلِّ شيء: يُعترَف بما قُدِّم حتى تاريخ الإلغاء** — بالمحرّك
        // نفسِه لا بقاعدةٍ ثانية (`ق-٤٦٩`). **فما بقي بعدها مقبوضٌ عن مدّةٍ لن
        // تُقدَّم**، وهو وحده موضوعُ السؤال.
        await recognizeDueWithinTx(tx, companyId, cancelDate, id);

        // ⭐ ثم **يُعكس استحقاقُ ما سيُسقَط** (ق-٩٢٥/②): قسطٌ حلّ بعد تاريخ الإلغاء ولم يُقبض
        // منه شيء ويُسقَط — تُردّ ذمّتُه وضريبتُه وإيرادُه كإشعارٍ دائن، **قبل** أن يُقرأ
        // الالتزامُ الباقي فلا يُسأل عن وجهةِ ما لم يعد التزاماً.
        if (dto.installmentDisposition === 'cancel') await reverseVoidedInstallmentsWithinTx(tx, companyId, accrualContract, cancelDate);

        const unearnedDetail = await unearnedDetailWithinTx(tx, id);
        const unearned = unearnedDetail.total;
        if (unearned > 0 && !dto.unearnedDisposition) {
          // ⛔ **ولا افتراض** (`ق-٤٧٢`): «كلا الخيارين يكذب على نصف الحالات».
          throw new BadRequestException(formatMessage(MSG.unearned_disposition_required, { amount: formatMoney(unearned) }));
        }

        const settlementAmount = await this.resolveSettlementAmount(tx, id, cancelDate, dto);
        // بنود التسوية (§16): التفصيل لا يغيّر المبلغ — يشرحه. مجموعٌ لا
        // يساويه = خطأ إدخال يُرفض قبل أي أثر.
        if (dto.settlementLines?.length) {
          // الجمع مدوَّرٌ عند كل تراكم والمقارنة بالتسامح المركزي — وكانت هنا
          // عتبةٌ مكتوبةٌ بيد على جمعٍ عائم، وهو ما كان يخترعه كل ملف لنفسه.
          if (!moneyEquals(sumMoney(dto.settlementLines.map((l) => l.amount)), settlementAmount)) {
            throw new BadRequestException(MSG.settlement_lines_mismatch);
          }
        }
        const settlementEntry =
          settlementAmount > 0
            ? await postContractSettlementEntry(tx, companyId, {
                unitId: existing.unitId,
                customerId: existing.customerId,
                contractNo: existing.no,
                tenant: existing.tenant,
                date: cancelDate,
                baseAmount: toBaseAmount(settlementAmount, Number(existing.exchangeRate)),
              })
            : null;

        const data: Prisma.ContractUncheckedUpdateInput = {
          status: 'CANCELLED',
          cancelDate,
          cancelReason: dto.cancelReason,
          ...(settlementAmount > 0 ? { settlementAmount, settlementEntryId: settlementEntry?.id ?? null } : {}),
        };
        // Shrinking `end` to the cancel date is what frees the unit up again —
        // UnitsService's status check only looks at contracts still covering
        // today (start <= today <= end).
        if (cancelDate < existing.end) data.end = cancelDate;

        await tx.contract.update({ where: { id }, data });

        if (dto.settlementLines?.length) {
          await tx.contractSettlementLine.createMany({
            data: dto.settlementLines.map((l) => ({ companyId, contractId: id, label: l.label, amount: l.amount })),
          });
        }

        if (dto.installmentDisposition === 'cancel') {
          await tx.installment.updateMany({
            where: { contractId: id, status: { in: ['DUE', 'LATE'] }, dueDate: { gte: cancelDate } },
            data: { status: 'CANCELLED' },
          });
        }

        // إقفالُ الالتزام الباقي — **بعد تعديل الأقساط**، فالمبلغُ مقروءٌ قبلها
        // ولا تغيّره حالةُ قسطٍ لم يُقبض منه شيء.
        if (unearned > 0 && dto.unearnedDisposition) {
          // ما **قُبض** عن مدّةٍ لن تُقدَّم: يُردّ أو يُحتجَز — كما كان (ق-٤٧٢).
          if (unearnedDetail.collected > 0) {
            await postDeferredSettlementEntry(tx, companyId, {
              unitId: existing.unitId,
              customerId: existing.customerId,
              contractNo: existing.no,
              tenant: existing.tenant,
              date: cancelDate,
              baseAmount: unearnedDetail.collected,
              disposition: dto.unearnedDisposition,
            });
          }
          // وما **استُحقّ ولم يُقبض** عن مدّةٍ بعد الإلغاء يتبع الوجهةَ نفسَها (ق-٩٢٧/٧):
          // `refund` ⇒ لا يدين به المستأجر فيُعكس؛ `income` ⇒ يبقى ذمّةً ويصير دخلاً آخر.
          await settleUnpaidUnearnedWithinTx(tx, companyId, accrualContract, unearnedDetail, cancelDate, dto.unearnedDisposition);
          // العدّادُ يُغلق: ما خرج من الالتزام لا يعود مرشَّحاً للتحوّل — وإلّا
          // رحّل الكنسُ الشهريّ التزاماً أُقفل بالفعل.
          await closeDeferredWithinTx(tx, id);
        }

        return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id }, include: CONTRACT_INCLUDE }));
        },
        ACCRUAL_TX_OPTIONS,
      ),
    );
  }

  /** Turns the chosen basis into a figure. Returns 0 for 'none', and for the
   * installment-derived bases when nothing is left unpaid — cancelling a fully
   * paid lease owes nothing. */
  private async resolveSettlementAmount(tx: Prisma.TransactionClient, contractId: string, cancelDate: Date, dto: CancelContractDto): Promise<number> {
    const basis = dto.settlementBasis ?? 'none';
    if (basis === 'none') return 0;
    if (basis === 'amount') return dto.settlementAmount ?? 0;

    const remaining = await tx.installment.findMany({
      where: { contractId, status: { in: ['DUE', 'LATE'] }, dueDate: { gte: cancelDate } },
      orderBy: { dueDate: 'asc' },
      ...(basis === 'installments' ? { take: dto.settlementInstallments } : {}),
      select: { amount: true },
    });

    return round2(remaining.reduce((sum, i) => sum + Number(i.amount), 0));
  }

  /**
   * **صندوقُه لا يُقاس هنا**: دفعةُ الإيجار مستندٌ بلا فرع (العقدُ والعقارُ لا يحملان فرعاً)، والمستندُ بلا فرعٍ خارجُ حكم
   * `ق-١٤٣٧` — وحركةُ البنك مرآتُها بلا فرعٍ كذلك.
   */
  recordPayment(companyId: string, contractId: string, dto: RecordPaymentDto) {
    // Retries on a journal-entry "no" collision from a concurrent payment.
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const contract = await tx.contract.findUnique({ where: { id: contractId } });
        if (!contract) throw new NotFoundException(MSG.contract_not_found);
        // ⚖️ لا قبضَ على عقدٍ ملغى ولا على قسطٍ ملغى (`ق-١١٩٨`/⑥ — مراجعةٌ خارجية): الإلغاءُ سوّى
        //    العقدَ وأسقط أقساطَه وعكس ما استُحقّ منها، فقبضٌ بعده يُرحّل استحقاقاً واعترافاً على
        //    التزامٍ لم يعد قائماً — كان يمرّ `201` (مقيس).
        if (contract.status === 'CANCELLED') throw new ConflictException(MSG.contract_cancelled_no_payment);

        let installment = null;
        if (dto.installmentId) {
          installment = await tx.installment.findUnique({ where: { id: dto.installmentId } });
          if (!installment || installment.contractId !== contractId) throw new NotFoundException(MSG.installment_not_found);
          if (installment.status === 'CANCELLED') throw new ConflictException(MSG.installment_cancelled_no_payment);
        }

        if (dto.method === 'BANK' && !dto.bankId) throw new BadRequestException(MSG.bank_required);
        // A bankId is honoured for **either** method: a bank transfer names the
        // account, and a cash receipt may name the branch cash box it landed in
        // (a cash box is just a Bank record pointed at a branch — CLAUDE.md 6.8).
        // Either way the balance moves and a transaction is written, so a
        // branch's cash box reflects real money rather than only the GL account.
        const bank = dto.bankId ? await tx.bank.findUnique({ where: { id: dto.bankId } }) : null;
        if (dto.bankId && !bank) throw new NotFoundException(MSG.bank_not_found);

        // A payment defaults to the contract's currency/rate, but can carry
        // its own — the gap between the rate the contract was priced at and
        // the one it settles at is the realized FX gain/loss.
        const payCurrency = dto.currency?.trim() || contract.currency;
        const payRate = dto.exchangeRate ?? Number(contract.exchangeRate);

        const discount = dto.discount ?? 0;
        const netAmount = round2(dto.amount - discount);
        if (netAmount <= 0) throw new BadRequestException(MSG.payment_net_must_be_positive);

        // ⭐ **مساران بحدٍّ معلَن** (ق-٩٢٦ · ق-٩٢٧/٢·٣): القبضُ على قسطٍ استحقاقيّ لعقدٍ مصنَّف
        // **يسدّد ذمّةً** (نقدٌ ⇄ `1200`) بعد أن يُستحقّ ما يلزم؛ وما سبق البناءَ (أساسُ
        // القبض) أو عقدٌ لم يُصنَّف بعد يبقى على مسار القبض القائم (نقدٌ ⇄ `4200`/`2400`)
        // — ويُحسب في الإقرار ويُعدّ باسمه، ولا يُرحَّل رجعياً.
        const accrualContract = await tx.contract.findUniqueOrThrow({ where: { id: contractId }, select: ACCRUAL_CONTRACT_SELECT });
        const treatment = treatmentOf(accrualContract);
        const accrualPath = treatment !== null && (installment === null || installment.accrualBasis === 'ACCRUAL');
        // الإجماليُّ بمقياس عملة العقد — قبل الخصم، فالخصمُ تعديلُ مقابلٍ لاحق.
        const baseExpected = payCurrency === contract.currency ? toBaseAmount(netAmount, Number(contract.exchangeRate)) : toBaseAmount(netAmount, payRate);
        const grossInContractCurrency = payCurrency === contract.currency ? dto.amount : round2(toBaseAmount(dto.amount, payRate) / (Number(contract.exchangeRate) || 1));

        let journalEntry: { id: string; no: string } | null = null;
        let earnedNow = 0;
        if (!accrualPath) {
          const posting = await postRentPaymentEntry(tx, companyId, {
            unitId: contract.unitId,
            customerId: contract.customerId,
            contractNo: contract.no,
            tenant: contract.tenant,
            date: new Date(dto.date),
            // فترةُ خدمةِ القسط — وبلا قسطٍ مرتبط لا فترةَ تُعرف (ق-٤٧٣).
            service:
              installment?.serviceStart && installment?.serviceEnd
                ? { start: installment.serviceStart, end: installment.serviceEnd }
                : null,
            baseNetAmount: toBaseAmount(netAmount, payRate),
            // Priced at the contract's rate — but only when the payment is in
            // the contract's own currency. A payment made in a different
            // currency altogether has no rate difference to recognize.
            baseExpectedAmount: baseExpected,
            // (ق-٢٧٧) الطرفُ المدين حسابُ البنك حين يُقبض الإيجارُ على بنك — لا `1000`.
            cashAccountId: bank?.accountId ?? null,
          });
          journalEntry = posting.entry;
          earnedNow = posting.earned;
          // قسطٌ استحقاقيٌّ لعقدٍ لم يُصنَّف قُبض على المسار القائم — فيُثبَّت على أساس القبض
          // كي لا يستحقّه الكنسُ ثانيةً بعد التصنيف (والصفُّ سابقٌ للبناء فالقيدُ يقبله).
          if (installment && installment.accrualBasis === 'ACCRUAL') {
            await tx.installment.update({ where: { id: installment.id }, data: { accrualBasis: 'LEGACY_COLLECTED' } });
          }
        }

        const payment = await tx.rentPayment.create({
          data: {
            contractId,
            installmentId: dto.installmentId,
            date: new Date(dto.date),
            amount: dto.amount,
            discount,
            netAmount,
            method: dto.method,
            // Records which money-holder took it — the bank account for a
            // transfer, or the branch cash box for a cash receipt.
            bankId: bank ? bank.id : undefined,
            notes: dto.notes,
            journalEntryId: journalEntry?.id,
            currency: payCurrency,
            exchangeRate: payRate,
            baseAmount: toBaseAmount(dto.amount, payRate),
            baseDiscount: toBaseAmount(discount, payRate),
            baseNetAmount: toBaseAmount(netAmount, payRate),
          } as unknown as Prisma.RentPaymentUncheckedCreateInput,
        });

        if (accrualPath && treatment) {
          // ① يُستحقّ ما يلزم — بعد كتابة الدفعة كي يقرأها المحرّك: نقطتُها المقدَّمة بقدرها
          //    إن سبقت الاستحقاق، أو باقي القسط إن حلّ، وما فاض عنه بقدره (م٢٠/٢ · ق-٩٢٤).
          const accrued = await accrueForPaymentWithinTx(
            tx,
            companyId,
            accrualContract,
            treatment,
            { id: payment.id, date: new Date(dto.date), grossInContractCurrency, installmentId: installment?.id ?? null },
            new Date(),
          );
          if (!accrued) throw new BadRequestException(MSG.rent_accrual_period_locked);

          // ② الخصمُ تعديلُ مقابلٍ (ق-٩٢٧/٦): يُنقص الذمّةَ والصافيَ والضريبة — لا يُهمَل.
          if (discount > 0) {
            const fresh = installment
              ? await tx.installment.findUniqueOrThrow({
                  where: { id: installment.id },
                  select: { id: true, baseAccruedGross: true, baseAccruedNet: true, baseAccruedTax: true, baseRecognized: true },
                })
              : null;
            const discountGross = payCurrency === contract.currency ? discount : round2(toBaseAmount(discount, payRate) / (Number(contract.exchangeRate) || 1));
            await discountAccrualWithinTx(tx, companyId, accrualContract, treatment, fresh, {
              date: new Date(dto.date),
              gross: discountGross,
              paymentId: payment.id,
            });
          }

          // ③ ثم القبضُ يسدّد الذمّة: نقدٌ بسعر يومه ⇄ `1200` بسعر العقد، وفرقُ الصرف على
          //    `5600` — كسند القبض على فاتورة (ق-٨٧١).
          const settlement = await postPartySettlementEntry(tx, companyId, {
            customerId: contract.customerId,
            side: 'RECEIVABLE',
            date: new Date(dto.date),
            memo: `قبض إيجار — عقد ${contract.no} — ${contract.tenant}`,
            baseCash: toBaseAmount(netAmount, payRate),
            baseSettled: baseExpected,
            cashAccountId: bank?.accountId ?? null,
            unitId: contract.unitId,
          });
          journalEntry = settlement;
          if (settlement) await tx.rentPayment.update({ where: { id: payment.id }, data: { journalEntryId: settlement.id } });
        }

        // مرآةُ البنك/الصندوق — بعد أن يُعرف قيدُ القبض على المسارَين (رقمُه في الحركة).
        if (bank) {
          await tx.bank.update({ where: { id: bank.id }, data: { balance: { increment: netAmount } } });
          await tx.bankTransaction.create({
            data: {
              bankId: bank.id,
              date: new Date(dto.date),
              description: `دفعة إيجار — عقد ${contract.no} — ${contract.tenant}`,
              amount: netAmount,
              source: 'rent',
              matched: !!journalEntry,
              entryNo: journalEntry?.no ?? null,
              currency: payCurrency,
              exchangeRate: payRate,
              baseAmount: toBaseAmount(netAmount, payRate),
            } as unknown as Prisma.BankTransactionUncheckedCreateInput,
          });
        }

        if (installment) {
          // **الحالةُ تُشتقّ من المقبوض مقابل الإجماليّ** — كما الفاتورة (`SPEC 4.5` · `ق-١٠٤٠`): لا ختمٌ ثابت.
          // الإجماليُّ بالضريبة من الدالّة نفسِها التي يقيس بها الاستحقاق (`rentAmountsOf`)، والقديمُ غيرُ
          // المصنَّف بمبلغه؛ والمقبوضُ مجموعُ دفعات القسط (الصافي + الخصم — فالخصمُ يُنقص المستحقَّ في
          // المحرّك `ق-٤٦٩`). و`PARTIAL` تتقدّم على `LATE` — المدفوعُ يتقدّم على كلِّ شيء — والتأخّرُ يُقرأ
          // من التاريخ. ⛔ ولا تُقرأ الحالةُ في الاستحقاق ولا في الأعمار ولا في الإقرار — بنقطة الضريبة وحدَها.
          const settledAgg = await tx.rentPayment.aggregate({ where: { installmentId: installment.id }, _sum: { netAmount: true, discount: true } });
          const settled = round2(Number(settledAgg._sum.netAmount ?? 0) + Number(settledAgg._sum.discount ?? 0));
          const gross = treatment ? rentAmountsOf(Number(installment.amount), treatment).gross : round2(Number(installment.amount));
          const derived = deriveSettlementStatus(settled, gross);
          const status = derived === 'PAID' ? 'PAID' : derived === 'PARTIAL' ? 'PARTIAL' : installment.status;
          await tx.installment.update({
            where: { id: installment.id },
            // العدّادُ يُزاد بما اعتُرف به **في قيد القبض القائم وحده** — وصفرٌ على مسار
            // الاستحقاق (الاعترافُ هناك من الاستحقاق لا من القبض) وحين تخطّى الترحيل (ق-٤٦٨).
            data: { status, baseRecognized: { increment: earnedNow } },
          });
        }

        return tx.rentPayment.findUniqueOrThrow({ where: { id: payment.id }, include: { installment: true } });
      }),
    );
  }

  private async validateContract(tx: Prisma.TransactionClient, unit: Unit, property: Property, start: Date, end: Date, tenantName: string, excludeContractId?: string): Promise<void> {
    if (start > end) throw new BadRequestException(MSG.contract_start_after_end);

    if (property.activityType === 'residential' && property.activitySubtype && unit.subtype && unit.subtype !== property.activitySubtype) {
      throw new ConflictException(MSG.unit_subtype_mismatch);
    }
    if (property.ownership === 'leased' && property.leaseStart && property.leaseEnd) {
      if (start < property.leaseStart || end > property.leaseEnd) {
        throw new ConflictException(MSG.contract_outside_head_lease);
      }
    }

    const overlapping = await tx.contract.findFirst({
      where: {
        unitId: unit.id,
        status: { not: 'CANCELLED' },
        ...(excludeContractId ? { id: { not: excludeContractId } } : {}),
        start: { lte: end },
        end: { gte: start },
      },
    });
    if (overlapping) throw new ConflictException(formatMessage(MSG.contract_dates_overlap, { contractNo: overlapping.no }));

    await expireStaleReservations(tx);
    const activeRsv = await activeReservationForUnit(tx, unit.id);
    if (activeRsv && activeRsv.name.trim() !== tenantName.trim()) {
      throw new ConflictException(
        formatMessage(MSG.reservation_name_mismatch, {
          name: activeRsv.name,
          expiryDate: activeRsv.expiryDate.toISOString().slice(0, 10),
        }),
      );
    }
  }

  /**
   * **العميلُ الذي يصحّ ربطُ عقدٍ به** (`ق-١٠٢٨`/①): يُقرأ **بمعرّفه** داخل سياق المنشأة — فالغائبُ
   * والمحذوفُ ناعماً (يُخفيه الامتداد) `404`، والمؤرشَفُ `409` تسمّي المخرج. لا بحثَ بالاسم ولا
   * إنشاءَ ضمنيّاً — بابُ الإنشاء وحدةُ العملاء (حارسُه `no-name-linking.spec.ts`).
   */
  private async linkableCustomer(tx: Prisma.TransactionClient, customerId: string): Promise<{ id: string; name: string; phone: string | null }> {
    const customer = await tx.customer.findUnique({ where: { id: customerId }, select: { id: true, name: true, phone: true, archived: true } });
    if (!customer) throw new NotFoundException(MSG.customer_not_found);
    if (customer.archived) throw new ConflictException(MSG.customer_archived_cannot_link);
    return { id: customer.id, name: customer.name, phone: customer.phone };
  }

  /**
   * Renews a lease: a **new** contract that carries the tenancy forward, linked
   * back to the one it replaces.
   *
   * Not an edit of the old contract. The expiring one keeps its own installments
   * and payment history — that record is what a dispute is settled against, and
   * stretching its dates would erase it.
   *
   * The renewal is created as UPCOMING when its term hasn't started yet: signed
   * and certain, unlike a DRAFT which may never happen. It becomes DUE by the
   * same date-driven refresh that already moves DUE to LATE.
   */
  renew(companyId: string, id: string, dto: RenewContractDto) {
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const source = await tx.contract.findUnique({ where: { id }, include: { unit: { include: { property: true } }, renewedTo: true } });
        if (!source) throw new NotFoundException(MSG.contract_not_found);
        if (source.status === 'CANCELLED') throw new ConflictException(MSG.contract_cancelled_cannot_renew);
        // The unique constraint would catch this too, but as an opaque 500.
        if (source.renewedTo) throw new ConflictException(formatMessage(MSG.contract_already_renewed, { contractNo: source.renewedTo.no }));

        const start = new Date(dto.start);
        const end = new Date(dto.end);
        if (start <= source.start) throw new BadRequestException(MSG.renewal_must_start_after_source);

        // The renewal is a real tenancy on the same unit, so it goes through
        // the same overlap and activity checks as any other contract. The
        // source itself is excluded: a renewal starting the day after it ends
        // is precisely the point, and an early renewal signed while the old
        // lease still runs must not collide with it either.
        // الطرفُ يُقرأ من بطاقة العميل لحظةَ التجديد — لا من نصّ العقد الأصل (`ق-١٠٢٨`/①⑥).
        const party = source.customerId ? await this.linkableCustomer(tx, source.customerId) : null;
        await this.validateContract(tx, source.unit, source.unit.property, start, end, party?.name ?? source.tenant, source.id);

        const value = dto.value ?? Number(source.value);
        const cycle = dto.cycle ?? source.cycle;
        // التجديد: بادئته من بداية العقد **الجديد** لا الأصل — فهو مستندٌ قائم
        // بذاته بمدّته الخاصة.
        const no = dto.no?.trim() || (await this.nextContractNumber(tx, companyId, start));
        if (dto.no?.trim()) {
          const taken = await tx.contract.findFirst({ where: { no } });
          if (taken) throw new ConflictException(formatMessage(MSG.contract_no_taken, { contractNo: no }));
        }

        const exchangeRate = Number(source.exchangeRate);
        const created = await tx.contract.create({
          data: {
            unitId: source.unitId,
            customerId: source.customerId,
            no,
            ejarNo: dto.ejarNo ?? null,
            tenant: party?.name ?? source.tenant,
            phone: party?.phone ?? source.phone,
            value,
            cycle,
            start,
            end,
            // The agreed deposit figure carries over as-is; whether the money
            // itself moves is `carryDeposits` below.
            deposit: source.deposit,
            services: source.services,
            furnished: source.furnished,
            // يُعلَنان في طلب التجديد لا يُورَّثان من الأصل (ق-٨٩١).
            classification: dto.classification,
            vatInclusion: dto.vatInclusion,
            status: start > new Date() ? 'UPCOMING' : 'DUE',
            currency: source.currency,
            exchangeRate,
            baseValue: toBaseAmount(value, exchangeRate),
            renewedFromId: source.id,
          } as unknown as Prisma.ContractUncheckedCreateInput,
        });

        const schedule = generateInstallmentSchedule(start, end, value, cycle);
        await tx.installment.createMany({
          data: schedule.map((s) => ({
            contractId: created.id,
            dueDate: s.dueDate,
            amount: s.amount,
            // كسابقتها في `create` — الفترةُ تُثبَّت لا تُشتقّ (ق-٤٧٥/②).
            serviceStart: s.serviceStart,
            serviceEnd: s.serviceEnd,
            currency: source.currency,
            exchangeRate,
            baseAmount: toBaseAmount(s.amount, exchangeRate),
          })) as unknown as Prisma.InstallmentUncheckedCreateInput[],
        });

        // Carrying a deposit forward posts **nothing**: the cash never moved,
        // and the liability is the same one — only the lease it secures
        // changed. Refunding and re-collecting it on paper would invent two
        // cash movements that never happened.
        if (dto.carryDeposits ?? true) {
          await tx.contractDeposit.updateMany({
            where: { contractId: source.id, status: 'HELD' },
            data: { contractId: created.id },
          });
        }

        // كسابقتها في `create`: ما حلّ من جدول التجديد يُستحقّ في اللحظة نفسِها.
        await accrueDueWithinTx(tx, companyId, new Date(), created.id);

        return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: created.id }, include: CONTRACT_INCLUDE }));
        },
        ACCRUAL_TX_OPTIONS,
      ),
    );
  }

  /**
   * Creates a condition-of-unit form as a draft.
   *
   * Nothing is posted: a handover is a record of state, not a financial event.
   * What it feeds is the *argument* over a later deposit deduction.
   */
  createHandover(companyId: string, contractId: string, dto: CreateHandoverDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const contract = await tx.contract.findUnique({ where: { id: contractId } });
      if (!contract) throw new NotFoundException(MSG.contract_not_found);

      const created = await tx.contractHandover.create({
        data: {
          contractId,
          type: dto.type,
          date: new Date(dto.date),
          electricityMeter: dto.electricityMeter?.trim() || null,
          waterMeter: dto.waterMeter?.trim() || null,
          notes: dto.notes?.trim() || null,
        } as unknown as Prisma.ContractHandoverUncheckedCreateInput,
      });

      await this.replaceHandoverItems(tx, created.id, dto.items);
      return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
    });
  }

  /** Draft only — see `assertHandoverEditable`. */
  updateHandover(companyId: string, contractId: string, handoverId: string, dto: UpdateHandoverDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const handover = await this.loadEditableHandover(tx, contractId, handoverId);

      await tx.contractHandover.update({
        where: { id: handover.id },
        data: {
          ...(dto.date !== undefined ? { date: new Date(dto.date) } : {}),
          ...(dto.electricityMeter !== undefined ? { electricityMeter: dto.electricityMeter.trim() || null } : {}),
          ...(dto.waterMeter !== undefined ? { waterMeter: dto.waterMeter.trim() || null } : {}),
          ...(dto.notes !== undefined ? { notes: dto.notes.trim() || null } : {}),
        },
      });

      if (dto.items) await this.replaceHandoverItems(tx, handover.id, dto.items);
      return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
    });
  }

  /**
   * Signs the form and freezes it.
   *
   * `signedAt` is the **server's** clock, never the caller's. A document whose
   * purpose is to be unarguable later cannot let one side choose the time it
   * claims to have been signed.
   */
  signHandover(companyId: string, contractId: string, handoverId: string, dto: SignHandoverDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const handover = await this.loadEditableHandover(tx, contractId, handoverId);

      await tx.contractHandover.update({
        where: { id: handover.id },
        data: { status: 'SIGNED', signedAt: new Date(), signedBy: dto.signedBy.trim() },
      });

      return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
    });
  }

  /** Deleting is a draft-only operation for the same reason editing is. */
  async deleteHandover(companyId: string, contractId: string, handoverId: string) {
    const { photoKeys, refreshed } = await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const handover = await this.loadEditableHandover(tx, contractId, handoverId);
      // صور المسودة تُحذف صفوفها بالتتالي — نلتقط مفاتيح كائناتها قبل ذلك
      // كي لا تبقى بايتاتها يتيمةً في التخزين (اكتشاف المراجعة).
      const photos = await tx.handoverPhoto.findMany({ where: { handoverId: handover.id }, select: { fileKey: true } });
      await tx.contractHandover.delete({ where: { id: handover.id } });
      const refreshed = this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
      return { photoKeys: photos.map((p) => p.fileKey), refreshed };
    });
    for (const key of photoKeys) {
      await this.files.remove(companyId, key).catch((e: Error) => this.logger.warn(`Could not delete handover photo object "${key}": ${e.message}`));
    }
    return refreshed;
  }

  /** The single gate every mutation goes through — one place to be wrong, and
   * one place to get right. */
  private async loadEditableHandover(tx: Prisma.TransactionClient, contractId: string, handoverId: string) {
    const handover = await tx.contractHandover.findUnique({ where: { id: handoverId } });
    if (!handover || handover.contractId !== contractId) throw new NotFoundException(MSG.handover_not_found);
    if (handover.status === 'SIGNED') throw new ConflictException(MSG.handover_signed_immutable);
    return handover;
  }

  private async replaceHandoverItems(tx: Prisma.TransactionClient, handoverId: string, items: HandoverItemDto[]) {
    // Photos hang off items by id, and this replace strategy re-creates every
    // item with a NEW id (onDelete: SetNull would silently orphan them —
    // caught in adversarial review). Remember which labels each photo pointed
    // at, and re-attach by label after the rebuild; a photo whose item was
    // truly removed degrades to a whole-form photo rather than vanishing.
    const linkedPhotos = await tx.handoverPhoto.findMany({
      where: { handoverId, itemId: { not: null } },
      select: { id: true, item: { select: { label: true } } },
    });

    await tx.handoverItem.deleteMany({ where: { handoverId } });

    if (items.length > 0) {
      await tx.handoverItem.createMany({
        data: items.map((i) => ({
          handoverId,
          label: i.label.trim(),
          condition: i.condition,
          note: i.note?.trim() || null,
        })) as unknown as Prisma.HandoverItemUncheckedCreateInput[],
      });
    }

    if (linkedPhotos.length > 0) {
      const fresh = await tx.handoverItem.findMany({ where: { handoverId }, select: { id: true, label: true } });
      const idByLabel = new Map(fresh.map((i) => [i.label, i.id]));
      for (const photo of linkedPhotos) {
        const newItemId = photo.item ? (idByLabel.get(photo.item.label) ?? null) : null;
        await tx.handoverPhoto.update({ where: { id: photo.id }, data: { itemId: newItemId } });
      }
    }
  }

  /**

  /** Collects a claim in cash. Partial collection isn't supported: a claim is a

  payClaim(companyId: string, contractId: string, claimId: string, dto: PayClaimDto) {

  }

  /** Writes a claim off. Reverses the income rather than deleting the claim —
   * it really was raised, and the audit trail has to show both facts. */
  waiveClaim(companyId: string, contractId: string, claimId: string, dto: WaiveClaimDto) {
    return this.settleClaim(companyId, contractId, claimId, 'WAIVED', new Date(dto.waivedAt), dto.reason.trim());
  }

  private settleClaim(companyId: string, contractId: string, claimId: string, status: 'PAID' | 'WAIVED', date: Date, waiveReason: string | null) {
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const claim = await tx.contractClaim.findUnique({ where: { id: claimId }, include: { contract: true } });
        if (!claim || claim.contractId !== contractId) throw new NotFoundException(MSG.claim_not_found);
        if (claim.status !== 'OPEN') throw new ConflictException(MSG.claim_already_settled);

        const posting = {
          unitId: claim.contract.unitId,
          customerId: claim.contract.customerId,
          contractNo: claim.contract.no,
          tenant: claim.contract.tenant,
          date,
          baseAmount: Number(claim.baseAmount),
          kindLabel: CLAIM_KIND_LABEL[claim.kind],
        };
        const entry = status === 'PAID' ? await postClaimPaymentEntry(tx, companyId, posting) : await postClaimWaiverEntry(tx, companyId, posting);

        await tx.contractClaim.update({
          where: { id: claimId },
          data: { status, settledAt: date, waiveReason, settlementEntryId: entry?.id ?? null },
        });

        return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
      }),
    );
  }

  /**
   * Records a deposit the tenant has actually handed over, and posts it as a
   * liability.
   *
   * `Contract.deposit` remains what was *agreed*; this is what was *received*.
   * The two are separate on purpose — a contract can be signed with a deposit
   * owed but not yet paid, and only the second belongs in the ledger.
   */
  recordDeposit(companyId: string, contractId: string, dto: RecordDepositDto) {
    // Retries on a journal-entry "no" collision from a concurrent posting.
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const contract = await tx.contract.findUnique({ where: { id: contractId } });
        if (!contract) throw new NotFoundException(MSG.contract_not_found);

        // A deposit defaults to the contract's currency and rate, since it is
        // part of the same deal, but may carry its own.
        const currency = dto.currency?.trim() || contract.currency;
        const exchangeRate = dto.exchangeRate ?? Number(contract.exchangeRate);
        const baseAmount = toBaseAmount(dto.amount, exchangeRate);

        // Only a deposit the company actually holds hits the ledger. One
        // blocked on Ejar is tracked here so it shows on the contract and can
        // be settled later, but the company neither received cash nor owes it
        // back — the platform is the custodian.
        const holder = dto.holder ?? 'COMPANY';
        const journalEntry =
          holder === 'COMPANY'
            ? await postDepositReceiptEntry(tx, companyId, {
                unitId: contract.unitId,
                customerId: contract.customerId,
                contractNo: contract.no,
                tenant: contract.tenant,
                date: new Date(dto.receivedAt),
                baseAmount,
                typeLabel: DEPOSIT_TYPE_LABEL[dto.type as DepositType],
              })
            : null;

        await tx.contractDeposit.create({
          data: {
            contractId,
            type: dto.type,
            holder,
            amount: dto.amount,
            currency,
            exchangeRate,
            baseAmount,
            receivedAt: new Date(dto.receivedAt),
            note: dto.note,
            journalEntryId: journalEntry?.id ?? null,
          } as unknown as Prisma.ContractDepositUncheckedCreateInput,
        });

        return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
      }),
    );
  }

  /**
   * Closes a deposit out: itemized deductions, and whatever is left goes back
   * to the tenant.
   *
   * Settling twice is refused rather than allowed to post a second entry — the
   * liability only exists once, and a repeat would credit cash the landlord
   * never paid.
   */
  settleDeposit(companyId: string, contractId: string, depositId: string, dto: SettleDepositDto) {
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const deposit = await tx.contractDeposit.findUnique({ where: { id: depositId }, include: { contract: true } });
        if (!deposit || deposit.contractId !== contractId) throw new NotFoundException(MSG.deposit_not_found);
        if (deposit.status === 'SETTLED') throw new ConflictException(MSG.deposit_already_settled);

        const held = Number(deposit.amount);
        // Rounded at every accumulation: two 2-decimal amounts can sum to a
        // float like 300.00000000000006 and spuriously exceed the deposit.
        const deducted = sumMoney(dto.deductions.map((d) => d.amount));
        if (deducted > held) {
          throw new BadRequestException(formatMessage(MSG.deductions_exceed_deposit, { held: formatMoney(held) }));
        }

        const rate = Number(deposit.exchangeRate);
        const sumOf = (target: 'receivable' | 'income') =>
          toBaseAmount(
            sumMoney(dto.deductions.filter((d) => deductionTarget(d.kind as DeductionKind) === target).map((d) => d.amount)),
            rate,
          );

        const refund = round2(held - deducted);

        const settlementEntry = await postDepositSettlementEntry(tx, companyId, {
          unitId: deposit.contract.unitId,
          customerId: deposit.contract.customerId,
          contractNo: deposit.contract.no,
          tenant: deposit.contract.tenant,
          date: new Date(dto.settledAt),
          // An Ejar-held deposit has no liability to clear — only what the
          // platform releases to the company gets posted.
          heldByCompany: deposit.holder === 'COMPANY',
          baseAmount: Number(deposit.baseAmount),
          againstReceivable: sumOf('receivable'),
          againstIncome: sumOf('income'),
          refund: toBaseAmount(refund, rate),
          typeLabel: DEPOSIT_TYPE_LABEL[deposit.type],
        });

        if (dto.deductions.length > 0) {
          await tx.depositDeduction.createMany({
            data: dto.deductions.map((d) => ({
              depositId,
              kind: d.kind,
              amount: d.amount,
              reason: d.reason,
            })) as unknown as Prisma.DepositDeductionUncheckedCreateInput[],
          });
        }

        await tx.contractDeposit.update({
          where: { id: depositId },
          data: {
            status: 'SETTLED',
            settledAt: new Date(dto.settledAt),
            refundAmount: refund,
            settlementEntryId: settlementEntry?.id ?? null,
          },
        });

        return this.withComputedFigures(await tx.contract.findUniqueOrThrow({ where: { id: contractId }, include: CONTRACT_INCLUDE }));
      }),
    );
  }

  /** `EJ-<year>-<seq>`, same max-based scheme as invoice/purchase/journal
   * numbering — see InvoicesService.nextInvoiceNumber for why count() would
   * be wrong here. */
  /**
   * بادئة رقم العقد من **`startDate`** — قرار المالك 2026-08-15.
   *
   * ولا تاريخ توقيع في نموذج العقد أصلاً (`create-contract.dto.ts` يحمل `start`
   * و`end` لا غير). والحجّة: **تاريخ العقد الفعلي الذي يعرفه الطرفان ويظهر في
   * المستند — ولا يعتمد على ساعة الخادم إطلاقاً.**
   *
   * ⚠️ و`signedAt` بندٌ مسجَّل بلا بدء (نماذج التسليم والتوقيع الرقمي). **وحين
   * يُضاف لا تنتقل البادئة إليه تلقائياً** — تغييرُ أساس الترقيم بعد إصدار أرقام
   * قرارٌ مستقل يُعاد فتحه، لا أثرٌ جانبي لحقلٍ جديد.
   */
  private async nextContractNumber(tx: Prisma.TransactionClient, companyId: string, startDate: Date): Promise<string> {
    // عدّادٌ ذرّيّ لا مسحُ سنة (ق-٣٢٨) — والبذرُ من أعلى قائمٍ مرّةً واحدة.
    return nextDocumentNumber(tx, companyId, startDate, { kind: 'EJ', pad: 4 }, (prefix) => highestSequenceOf(tx.contract, prefix));
  }

  /** Lazily flips DUE -> LATE for any installment whose due date has passed
   * — matches the mockup's `refreshInstallmentStatuses`, called on every
   * read rather than via a background job. */
  private async refreshInstallmentStatuses(tx: Prisma.TransactionClient, contractId?: string): Promise<void> {
    await tx.installment.updateMany({
      where: { status: 'DUE', dueDate: { lt: new Date() }, ...(contractId ? { contractId } : {}) },
      data: { status: 'LATE' },
    });

    // An UPCOMING renewal becomes live the day its term starts. Driven by the
    // same lazy read-time refresh as the line above rather than a scheduled
    // job — a contract that only turns active when someone happens to look at
    // it is still correct, because nothing reads it without going through here.
    await tx.contract.updateMany({
      where: { status: 'UPCOMING', start: { lte: new Date() }, ...(contractId ? { id: contractId } : {}) },
      data: { status: 'DUE' },
    });
  }

  /**
   * Paginated, slim list — contract row + unit/property only, never the full
   * sub-tree. The old shape (every contract with all installments, payments,
   * deposits, claims, key money and handovers) grew linearly with portfolio
   * size and blew the tenant transaction's 5s timeout at ~1,300 contracts
   * (~12 MB of JSON). The detail modal loads one contract's sub-tree via
   * findOne, which keeps CONTRACT_INCLUDE.
   *
   * Statuses stay mockup-faithful: the same lazy refresh runs first, and each
   * row's `status` is the stored, freshly-refreshed value — nothing is
   * computed per row.
   */
  private async listRefreshed(tx: Prisma.TransactionClient, query: ListContractsQueryDto) {
    await this.refreshInstallmentStatuses(tx);

    const page = query.page ?? 1;
    const pageSize = query.pageSize ?? 25;
    const q = query.search?.trim();
    const where: Prisma.ContractWhereInput = q
      ? {
          OR: [
            { tenant: { contains: q, mode: 'insensitive' } },
            { no: { contains: q, mode: 'insensitive' } },
            { unit: { unitNo: { contains: q, mode: 'insensitive' } } },
          ],
        }
      : {};

    // Sequential on purpose: an interactive transaction runs on a single
    // connection, so concurrent queries on `tx` would only race for it.
    const total = await tx.contract.count({ where });
    const rows = await tx.contract.findMany({
      where,
      include: { unit: { include: { property: true } }, customer: { select: { id: true, name: true, phone: true, archived: true } } },
      // ترتيبٌ تامّ لا جزئيّ (`ق-٣٥٦`) — `start` بدقّة اليوم فيتعادل، و`id` يحسم.
      orderBy: [{ start: 'desc' }, { createdAt: 'desc' }, { id: 'desc' }],
      skip: (page - 1) * pageSize,
      take: pageSize,
    });
    // الاسمُ والهاتفُ من بطاقة العميل عند القراءة (`ق-١٠٢٨`/⑥) — والبحثُ أعلاه على الذاكرة المشتقّة
    // التي تُحدَّث مع تعديل العميل، فلا تتخلّف إلا لعقودٍ افترقت قبل هذا البناء.
    const items = rows.map(withPartyFromCustomer);

    return { items, total, page, pageSize };
  }
}

import { Type } from 'class-transformer';
import { IsArray, IsIn, IsInt, IsNotEmpty, IsNumber, IsOptional, IsPositive, IsString, Min, ValidateIf, ValidateNested } from 'class-validator';

const SETTLEMENT_BASES = ['none', 'amount', 'installments', 'all'] as const;

/** بند من تفصيل التسوية: «شهر إشعار: 3000» · «خصم صيانة: -450». */
export class SettlementLineDto {
  @IsString()
  @IsNotEmpty()
  label!: string;

  /** موجب = على المستأجر، سالب = له — والمجموع يجب أن يساوي مبلغ التسوية. */
  @IsNumber()
  amount!: number;
}

export class CancelContractDto {
  @IsString()
  @IsNotEmpty()
  cancelDate!: string;

  @IsString()
  @IsNotEmpty()
  cancelReason!: string;

  /** 'keep' leaves not-yet-due installments owed as-is; 'cancel' voids every
   * unpaid installment due on/after the cancel date — matches the mockup's
   * cancel-contract dialog. */
  @IsIn(['keep', 'cancel'])
  installmentDisposition!: 'keep' | 'cancel';

  /**
   * How the early-termination settlement is worked out:
   *
   * - `none` — the parties agreed on nothing, so nothing is posted;
   * - `amount` — an explicitly negotiated figure;
   * - `installments` — the next N unpaid installments' worth;
   * - `all` — everything still unpaid from the cancel date onwards.
   *
   * The last two are conveniences that compute an amount; only the number they
   * arrive at is stored, because the installments themselves may be voided in
   * the same call.
   */
  @IsOptional()
  @IsIn(SETTLEMENT_BASES)
  settlementBasis?: (typeof SETTLEMENT_BASES)[number];

  @ValidateIf((o) => o.settlementBasis === 'amount')
  @IsNumber()
  @IsPositive()
  settlementAmount?: number;

  /** تفصيل التسوية بنوداً (§16) — اختياري؛ عند وجوده يُدقَّق مجموعه ضد
   * المبلغ المحسوب ويُخزَّن سطراً سطراً. */
  @IsOptional()
  @IsArray()
  @ValidateNested({ each: true })
  @Type(() => SettlementLineDto)
  settlementLines?: SettlementLineDto[];

  @ValidateIf((o) => o.settlementBasis === 'installments')
  @IsInt()
  @Min(1)
  settlementInstallments?: number;

  /**
   * ⚖️ **وجهةُ المقبوض عن مدّةٍ لم تُقدَّم — ولا قيمةَ افتراضية** (`ق-٤٧٢`).
   *
   * - `refund` — **يُردّ للمستأجر**: نقدٌ يخرج مقابلَ الالتزام.
   * - `income` — **يُحتجَز بشرط العقد**: دخلٌ آخر (`4300`).
   *
   * **⛔ ولا يذهب إلى `4200` بحال**: إيرادُ الإيجار يُكتسب بتأجير العقارِ مدّةً،
   * **وهذه مدّةٌ لا يشغلها المستأجر** — وهي حجّةُ `4300` المنصوصةُ في
   * `auto-posting.util.ts` منذ تسوية الإلغاء.
   *
   * ⚠️ **وصمتُه يمنع الإلغاء** حين يوجد التزامٌ فعليّ — **فكلا الخيارين يكذب
   * على نصف الحالات، ولا افتراضَ آمن** (حكمُ المالك). **ولا يُطلَب حين لا
   * التزام**، فسؤالٌ بلا موضوعٍ يُضغَط بلا قراءة.
   *
   * 📌 **ولماذا حقلٌ مستقلٌّ لا `settlementBasis`**: ذاك يحسب **ما على
   * المستأجر**، وهذا يقرّر **وجهةَ ما قُبض منه** — كمّيتان مختلفتان. **وحقلٌ
   * يُستعمل في غير ما وُضع له يُنتج رقماً يُقرأ صحيحاً وهو كاذب** (`ق-٤٦١`).
   */
  @IsOptional()
  @IsIn(['refund', 'income'])
  unearnedDisposition?: 'refund' | 'income';
}

// FILE: apps/api/src/contracts/handover-photos.service.ts

import { ConflictException, Injectable, Logger, NotFoundException } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { TenantPrismaService } from '../common/tenant/tenant-prisma.service';
import { FilesService } from '../files/files.service';
import { MSG } from '../common/i18n/messages';

const PHOTO_SELECT = {
  id: true,
  handoverId: true,
  itemId: true,
  filename: true,
  size: true,
  mimeType: true,
  note: true,
  uploadedAt: true,
} as const;

/**
 * صور نموذج الاستلام/التسليم (16.15): على النموذج كله أو على بندٍ بعينه
 * (خدشُ البابِ صورتُه على بند الباب).
 *
 * القاعدة الحاكمة: الصور **جزء مما وُقِّع عليه** — بعد التوقيع لا تُضاف صورة
 * ولا تُحذف، بنفس صرامة بنود النموذج نفسها (handover_signed_immutable).
 * العرض والتنزيل يبقيان مفتوحين دائماً.
 */
@Injectable()
export class HandoverPhotosService {
  private readonly logger = new Logger(HandoverPhotosService.name);

  constructor(
    private readonly tenantPrisma: TenantPrismaService,
    private readonly files: FilesService,
  ) {}

  list(companyId: string, handoverId: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const handover = await tx.contractHandover.findUnique({ where: { id: handoverId }, select: { id: true } });
      if (!handover) throw new NotFoundException(MSG.handover_not_found);
      return tx.handoverPhoto.findMany({ where: { handoverId }, select: PHOTO_SELECT, orderBy: { uploadedAt: 'asc' } });
    });
  }

  async add(
    companyId: string,
    handoverId: string,
    file: { originalname: string; mimetype: string; buffer: Buffer; size: number },
    opts: { itemId?: string; note?: string },
  ) {
    const stored = await this.files.upload(companyId, file);
    try {
      return await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const handover = await tx.contractHandover.findUnique({ where: { id: handoverId }, select: { id: true, status: true } });
        if (!handover) throw new NotFoundException(MSG.handover_not_found);
        // ما وُقِّع قُفِل — إضافة صورة بعد التوقيع تزوير للنموذج.
        if (handover.status === 'SIGNED') throw new ConflictException(MSG.handover_signed_immutable);

        if (opts.itemId) {
          const item = await tx.handoverItem.findFirst({ where: { id: opts.itemId, handoverId }, select: { id: true } });
          if (!item) throw new NotFoundException(MSG.handover_item_not_found);
        }

        const created = await tx.handoverPhoto.create({
          // companyId omitted on purpose — injected by the tenant-scope extension.
          data: {
            handoverId,
            itemId: opts.itemId ?? null,
            fileKey: stored.key,
            filename: stored.filename,
            size: stored.size,
            mimeType: stored.mimeType,
            note: opts.note,
          } as unknown as Prisma.HandoverPhotoUncheckedCreateInput,
          select: PHOTO_SELECT,
        });
        // سدّ سباق التوقيع المتزامن: لو وُقِّع النموذج بين فحصنا والإدراج،
        // قراءةٌ ثانية داخل نفس المعاملة تراه فنتراجع كلياً (rollback).
        const still = await tx.contractHandover.findUnique({ where: { id: handoverId }, select: { status: true } });
        if (still?.status === 'SIGNED') throw new ConflictException(MSG.handover_signed_immutable);
        return created;
      });
    } catch (e) {
      await this.files.remove(companyId, stored.key).catch((err: Error) => this.logger.warn(`Could not roll back orphaned upload "${stored.key}": ${err.message}`));
      throw e;
    }
  }

  downloadUrl(companyId: string, photoId: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const photo = await tx.handoverPhoto.findUnique({ where: { id: photoId }, select: { fileKey: true } });
      if (!photo) throw new NotFoundException(MSG.file_not_found);
      return this.files.downloadUrl(companyId, photo.fileKey);
    });
  }

  async remove(companyId: string, photoId: string): Promise<void> {
    const fileKey = await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const photo = await tx.handoverPhoto.findUnique({ where: { id: photoId }, select: { fileKey: true, handoverId: true, handover: { select: { status: true } } } });
      if (!photo) throw new NotFoundException(MSG.file_not_found);
      if (photo.handover.status === 'SIGNED') throw new ConflictException(MSG.handover_signed_immutable);
      await tx.handoverPhoto.delete({ where: { id: photoId } });
      // سدّ سباق التوقيع المتزامن: لو وُقِّع بين فحصنا والحذف، قراءةٌ ثانية
      // داخل نفس المعاملة تراه فيتراجع الحذف كلياً وتبقى الصورة.
      const still = await tx.contractHandover.findUnique({ where: { id: photo.handoverId }, select: { status: true } });
      if (still?.status === 'SIGNED') throw new ConflictException(MSG.handover_signed_immutable);
      return photo.fileKey;
    });
    await this.files.remove(companyId, fileKey).catch((e: Error) => this.logger.warn(`Could not delete photo object "${fileKey}": ${e.message}`));
  }
}

const CYCLE_STEP_MONTHS: Record<string, number> = { monthly: 1, quarterly: 3, semiannual: 6, annual: 12 };

/**

 *

/** اليومُ السابق — والحسابُ بالتقويم المحلّي كبقيّة هذا الملفّ. */
function dayBefore(d: Date): Date {
  const out = new Date(d);
  out.setDate(out.getDate() - 1);
  return out;
}

/** Splits `value` evenly across installments stepped by `cycle`, with any
 * rounding remainder absorbed into the last installment — matches the
 * mockup's `generateInstallments`. Installment count is derived from the
 * contract's total duration in months (approximated via a 30.44-day month,
 * same as the mockup) divided by the cycle's step. */
export function generateInstallmentSchedule(start: Date, end: Date, value: number, cycle: string): ScheduledInstallment[] {
  const stepMonths = CYCLE_STEP_MONTHS[cycle] ?? 1;
  const totalDays = Math.max(1, (end.getTime() - start.getTime()) / (1000 * 60 * 60 * 24));
  const totalMonths = Math.max(1, Math.round(totalDays / 30.44));
  const count = Math.max(1, Math.round(totalMonths / stepMonths));

  // التوزيع من المحرّك: المجموع يساوي قيمة العقد بالضبط، والفرق على القسط
  // الأخير وحده. وكان هنا `Math.floor((value / count) * 100) / 100` — فعقدٌ بـ
  // `3.03` على ثلاثة يعطي `1.00 · 1.00 · 1.03` بدل `1.01` ثلاثاً، لأن
  // `3.03/3 × 100` يساوي `100.99999999999999`. راجع `common/money/allocation.ts`.
  const amounts = allocateEvenly(value, count);

  const dueDates: Date[] = [];
  const cursor = new Date(start);
  for (let i = 0; i < count; i++) {
    dueDates.push(new Date(cursor));
    cursor.setMonth(cursor.getMonth() + stepMonths);
  }

  return dueDates.map((dueDate, i) => {
    const next = dueDates[i + 1];
    // آخرُ قسطٍ ينتهي بنهاية العقد لا بخطوةٍ إضافية — **كما يبتلع الأخيرُ باقيَ
    // التقريب في المبلغ**، يبتلع باقيَ المدة. فعددُ الأقساط مقرَّبٌ (`round`)،
    // وخطوةٌ سادسةٌ على عقدٍ مدّتُه خمسةٌ ونصف تمتدّ خارج العقد.
    const periodEnd = next ? dayBefore(next) : new Date(end);
    return {
      dueDate,
      amount: amounts[i],
      serviceStart: new Date(dueDate),
      // فترةٌ مقلوبة لا تُكتب: عقدٌ فرَط تقريبُه يعطي بدايةً بعد نهايته،
      // **ويومٌ واحد أصدقُ من مدىً سالبٍ يقسم عليه المحرّك**.
      serviceEnd: periodEnd < dueDate ? new Date(dueDate) : periodEnd,
    };
  });
}

/**
 * **كنسُ الاستحقاق اليوميّ — نداءٌ للمحرّك لا قاعدةٌ ثانية** (`ق-٩٢٧/٥`).
 *
 * الاستحقاقُ يُرحَّل عند الإنشاء والتجديد والقبض وقبل قراءة الإقرار في مساراتها؛ وهذا

@Injectable()

    private readonly tenantPrisma: TenantPrismaService,
    private readonly prisma: PrismaService,
    private readonly recognition: RentRecognitionService,
  ) {}

  /** يُرحّل كلَّ ما حلّ حتى `asOf` لمنشأةٍ واحدة. */
  async runForCompany(companyId: string, asOf: Date = new Date()): Promise<AccrualSweepResult> {
    return this.tenantPrisma.runInTenantContext(companyId, (tx) => accrueDueWithinTx(tx, companyId, asOf), BATCH_TX_OPTIONS);
  }

  @Cron('0 1 * * *', { timeZone: SCHEDULE_TIME_ZONE })
  async runForAllCompanies(): Promise<{ companies: number; accrued: number }> {

    const companies = await this.prisma.company.findMany({ where: { status: 'ACTIVE' }, select: { id: true } });

    let accrued = 0;
    for (const c of companies) {
      try {
        const result = await this.runForCompany(c.id);
        accrued += result.accrued;
        if (result.blockedByPeriodLock > 0) {
          this.logger.warn(`استحقاقُ الإيجار: ${result.blockedByPeriodLock} قسطاً حجبته فترةٌ مقفلة في المنشأة ${c.id}`);

        this.logger.error(`استحقاقُ الإيجار أخفق للمنشأة ${c.id}`, e instanceof Error ? e.stack : String(e));
      }
    }

    if (accrued > 0) this.logger.log(`استحقاقُ الإيجار: ${accrued} صفّاً عبر ${companies.length} منشأة`);
    return { companies: companies.length, accrued };
  }
}

import { Injectable, Logger } from '@nestjs/common';

/**
 * **تحوّلُ الإيراد المؤجَّل — الطرفُ الثاني من القاعدة** (`ق-٤٦٨` · `ق-٤٧٥/①`).
 *
 * القبضُ يُنشئ التزاماً بما لم يُقدَّم بعد، **وهذه تُخرجه إيراداً شهراً بشهر**.
 * **ولا يُبنى المسارُ بأحدهما**: التزامٌ لا يتحوّل يُنتج ميزانيةً كاذبة كما
 * يُنتجه الاعترافُ الفوريّ.
 *
 * ## بالاستحقاق لا بالنبضة
 *
 * لا تسأل «ما حصّةُ هذا الشهر؟» بل **«ما الذي استُحقّ حتى اليوم ولم يُعترَف
 * به؟»** — كـ`outbox.dispatchDue`. **وأثرُ ذلك ثلاثة**:
 *
 *   ① **شهرٌ فات لتعطّلٍ يُعوَّض في التشغيل التالي — بتاريخه هو** لا بتاريخ
 *      التعافي، فيُقرأ في شهره من القائمة.
 *   ② **وتشغيلٌ مكرَّرٌ يُرحّل صفراً**: الهدفُ تراكميّ، فما اعتُرف به لا يُعاد.
 *   ③ **ولا انحرافَ تقريبٍ عبر اثني عشر شهراً**: المُرحَّلُ فرقُ تراكمَين.
 *
 * ## حدُّها المعلَن
 *
 * ⚠️ **`ScheduleModule` مُعطَّلٌ تحت الاختبار** (`ق-٢٩٢`) — **فلا اختبارَ يعبر
 * تشغيلَها المجدوَل**، وحارسُ `scheduler-registered.spec.ts` يثبت التسجيلَ لا
 * السلوك. **فالمنطقُ كلُّه في `runForCompany` وتُثبَت به**، ولا يحمل `@Cron`
 * إلا نداءً.
 *
 * ⚠️ **ومثيلان من الخادم يشغّلانها مرّتين** — **وهو غيرُ ضارٍّ بالبناء**: الهدفُ
 * التراكميّ يجعل الثانيةَ لا تُرحّل شيئاً، تماماً كخاصّية `escalation`.
 */
/** مهلةُ معاملةِ الكنس الدفعيّ (الاعترافُ والاستحقاق) — يقرأ منها `RentAccrualService` أيضاً. */
export const BATCH_TX_OPTIONS = { timeout: 600_000, maxWait: 30_000 };

@Injectable()
export class RentRecognitionService {
  private readonly logger = new Logger(RentRecognitionService.name);

  constructor(
    private readonly tenantPrisma: TenantPrismaService,
    private readonly prisma: PrismaService,
  ) {}

  /**
   * يُرحّل كلَّ ما استُحقّ حتى `asOf` لمنشأةٍ واحدة.
   *
   * **الالتزامُ لقسطٍ = المقبوضُ − `baseRecognized`.** ولكلِّ موعدِ اعترافٍ
   * هدفٌ تراكميّ، **والمُرحَّلُ فرقُه عمّا سبق** — فلا يُعاد ولا يُنسى.
   */
  async runForCompany(companyId: string, asOf: Date = new Date()): Promise<{ entries: number; recognized: number }> {
    // كنسٌ دفعيّ لا طلبٌ: قيدٌ لكلِّ شهرٍ لكلِّ قسطٍ قائم — فمهلتُه مهلةُ دفعةٍ لا خمسُ ثوانٍ.
    return this.tenantPrisma.runInTenantContext(companyId, (tx) => recognizeDueWithinTx(tx, companyId, asOf), BATCH_TX_OPTIONS);
  }

  /** عقدٌ واحد — للاختبارات وللمسارات التي تعرف عقدَها. */
  async runForContract(companyId: string, contractId: string, asOf: Date = new Date()): Promise<{ entries: number; recognized: number }> {
    return this.tenantPrisma.runInTenantContext(companyId, (tx) => recognizeDueWithinTx(tx, companyId, asOf, contractId), BATCH_TX_OPTIONS);
  }

  /**
   * أوّلَ كلِّ شهرٍ عند الثانية فجراً — **و`asOf` يومُ التشغيل نفسُه**، فمواعيدُ
   * الاعتراف تُشتقّ منه: آخرُ شهرٍ اكتمل وما فاته قبلَه.
   *
   * وفشلُ منشأةٍ لا يوقف البقيّة — كسابقتها في `escalation`.
   */
  @Cron('0 2 1 * *', { timeZone: SCHEDULE_TIME_ZONE })
  async runForAllCompanies(): Promise<{ companies: number; entries: number }> {
    // العميلُ غيرُ المقيَّد عمداً: هذا الموضعُ يعبر المستأجرين ليدخل سياقَ كلٍّ.
    const companies = await this.prisma.company.findMany({ where: { status: 'ACTIVE' }, select: { id: true } });

    let entries = 0;
    for (const c of companies) {
      try {
        const result = await this.runForCompany(c.id);
        entries += result.entries;
      } catch (e) {
        this.logger.error(`تحوّلُ الإيراد المؤجَّل أخفق للمنشأة ${c.id}`, e instanceof Error ? e.stack : String(e));
      }
    }

    // لا يُسجَّل إلا ما جرى فعلاً — سطرٌ شهريٌّ يقول «صفر» يُهجَر فلا يُقرأ.
    if (entries > 0) this.logger.log(`الإيراد المؤجَّل: ${entries} قيداً عبر ${companies.length} منشأة`);
    return { companies: companies.length, entries };
  }
}

/**
 * **المحرّكُ نفسُه داخل معاملةٍ قائمة** — كي يستعمله الإلغاءُ بلا معاملةٍ ثانية.
 *
 * ⭐ **ومصدرٌ واحدٌ للقاعدة** (`ق-٤٦٩`): الكنسُ الشهريّ والإلغاءُ يمرّان بهذه
 * الدالّة عينِها — **فلا تتباعد قاعدتان تفعلان الشيءَ نفسه**.
 *
 * `contractId` يقصُرها على عقدٍ واحد؛ وبدونه تكنس المنشأة كلَّها.
 */
export async function recognizeDueWithinTx(
  tx: Prisma.TransactionClient,
  companyId: string,
  asOf: Date,
  contractId?: string,
): Promise<{ entries: number; recognized: number }> {
  // مرشّحون: أقساطٌ بدأت خدمتُها، وأساسُها (المستحقُّ أو المقبوض) أكثرُ ممّا اعتُرف به —
  // **يُرشَّحون في القاعدة لا في العميل** (ق-٧٠٦ بروحه): كان الكنسُ يحمّل كلَّ قسطٍ بدأت خدمتُه
  // بدفعاته (عشراتُ الآلاف على التطوير) ليُخرج بضعةً، فيقترب من مهلة معاملة العزل (ق-٧٠٤).
  const ids = await tx.$queryRaw<{ id: string }[]>`
    SELECT i.id
    FROM "contract_installments" i
    LEFT JOIN (
      SELECT p."installmentId", SUM(p."baseNetAmount") AS collected
      FROM "rent_payments" p
      WHERE p."companyId" = ${companyId} AND p."installmentId" IS NOT NULL
      GROUP BY p."installmentId"
    ) p ON p."installmentId" = i.id
    WHERE i."companyId" = ${companyId}
      AND i."serviceStart" IS NOT NULL AND i."serviceStart" <= ${asOf}
      AND i."serviceEnd" IS NOT NULL
      ${contractId ? Prisma.sql`AND i."contractId" = ${contractId}` : Prisma.empty}
      AND (CASE WHEN i."accrualBasis" = 'ACCRUAL' THEN i."baseAccruedNet" ELSE COALESCE(p.collected, 0) END) > i."baseRecognized"
  `;
  const candidates = await tx.installment.findMany({
    where: { id: { in: ids.map((r) => r.id) } },
    select: {
      id: true,
      serviceStart: true,
      serviceEnd: true,
      baseRecognized: true,
      accrualBasis: true,
      baseAccruedNet: true,
      contract: { select: { no: true, tenant: true, unitId: true, customerId: true } },
      payments: { select: { baseNetAmount: true } },
    },
  });

  let entries = 0;
  let recognized = 0;

  for (const i of candidates) {
    // الأساسُ: المستحقُّ صافياً لقسطٍ على أساس الاستحقاق، والمقبوضُ لما سبق البناء (ق-٩٢٧/٢).
    const basis = recognitionBasis(i);
    if (basis <= 0) continue;

    const period = { start: i.serviceStart!, end: i.serviceEnd! };
    let already = Number(i.baseRecognized);
    if (already >= basis) continue;

    for (const step of recognitionSchedule(period, basis, asOf)) {
      const delta = round2(step.cumulative - already);
      if (delta <= 0) continue;

      const entry = await postDeferredRecognitionEntry(tx, companyId, {
        unitId: i.contract.unitId,
        customerId: i.contract.customerId,
        contractNo: i.contract.no,
        tenant: i.contract.tenant,
        date: step.date,
        baseAmount: delta,
        // الحدُّ يُعلَن في وصف القيد نفسِه لا في وثيقة (ق-٤٨٠).
        periodLabel: recognitionShareLabel(step.date, period.end),
      });
      // تخطٍّ صامتٌ لنقص حسابٍ لا يُسجَّل اعترافاً: العدّادُ يتبع الدفترَ
      // لا العكس، **وإلّا ضاع التزامٌ لم يخرج من الميزانية أبداً**.
      if (!entry) break;

      already = round2(already + delta);
      entries += 1;
      recognized = round2(recognized + delta);
    }

    if (already !== Number(i.baseRecognized)) {
      await tx.installment.update({ where: { id: i.id }, data: { baseRecognized: new Prisma.Decimal(already) } });
    }
  }

  return { entries, recognized };
}

/**
 * **الالتزامُ الباقي على عقدٍ** = المقبوضُ − المعترَفُ به، لكلِّ أقساطه.
 *
 * يُقرأ **بعد** `recognizeDueWithinTx` حتى تاريخ الإلغاء — **فما بقي بعدها
 * مقبوضٌ عن مدّةٍ لن تُقدَّم**، وهو وحده موضوعُ السؤال (`ق-٤٧٢`).
 */
export async function outstandingDeferredWithinTx(tx: Prisma.TransactionClient, contractId: string): Promise<number> {
  const installments = await tx.installment.findMany({
    where: { contractId },
    select: { baseRecognized: true, accrualBasis: true, baseAccruedNet: true, payments: { select: { baseNetAmount: true } } },
  });

  return round2(installments.reduce((sum, i) => sum + Math.max(0, recognitionBasis(i) - Number(i.baseRecognized)), 0));
}

/**
 * **أساسُ الاعتراف لقسطٍ** (`ق-٩٢٧/٢`): **المستحقُّ صافياً** على أساس الاستحقاق (`ق-٩١٧`:
 * الاستحقاقُ ينقل لحظةَ القيد ولا يغيّر مبدأَ الاعتراف)، **والمقبوضُ** لقسطٍ سبق البناءَ
 * (`LEGACY_COLLECTED`) — دالّةٌ واحدة يقرأ منها الكنسُ والإلغاءُ والإقفال.
 */
export function recognitionBasis(i: {
  accrualBasis: 'ACCRUAL' | 'LEGACY_COLLECTED';
  baseAccruedNet: Prisma.Decimal | number;
  payments: { baseNetAmount: Prisma.Decimal | number }[];
}): number {
  if (i.accrualBasis === 'ACCRUAL') return round2(Number(i.baseAccruedNet));
  return round2(i.payments.reduce((sum, p) => sum + Number(p.baseNetAmount), 0));
}

/**
 * **يُغلق العدّاد بعد إقفال الالتزام بقيدٍ واحد** — فلا يعود الكنسُ الشهريّ
 * يرى التزاماً خرج بالفعل، **فيُرحّله ثانيةً بلا مقابل**.
 *
 * ⚠️ **والعدّادُ يقول «ما خرج من الالتزام» لا «ما صار إيراداً»**: الخارجُ قد
 * يكون إيراداً شهرياً، **أو ردّاً للمستأجر، أو دخلاً آخر عند الإلغاء** — وهي
 * وجهاتٌ ثلاثٌ لالتزامٍ واحد.
 */
export async function closeDeferredWithinTx(tx: Prisma.TransactionClient, contractId: string): Promise<void> {
  const installments = await tx.installment.findMany({
    where: { contractId },
    select: { id: true, baseRecognized: true, accrualBasis: true, baseAccruedNet: true, payments: { select: { baseNetAmount: true } } },
  });

  for (const i of installments) {
    const basis = recognitionBasis(i);
    if (basis <= Number(i.baseRecognized)) continue;
    await tx.installment.update({ where: { id: i.id }, data: { baseRecognized: new Prisma.Decimal(basis) } });
  }
}

import { ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { Prisma, Unit } from '@prisma/client';
import { TenantPrismaService } from '../common/tenant/tenant-prisma.service';
import { expireStaleReservations } from '../reservations/reservations.service';
import { BulkCreateUnitsDto } from './dto/bulk-create-units.dto';
import { CreateUnitDto } from './dto/create-unit.dto';
import { UpdateUnitDto } from './dto/update-unit.dto';
import { formatMessage, MSG } from '../common/i18n/messages';
import { PagedQueryDto } from '../common/dto/paged-query.dto';
import { pageArray } from '../common/pagination/paged.util';

/** حقولُ البحث المُعلَنة (`ق-١٢٣٦`) — رقمُ الوحدة ونوعُها وطابقُها. */
const UNIT_SEARCH_FIELDS = ['unitNo', 'type', 'subtype', 'floor'] as const;

export type UnitStatus = 'vacant' | 'rented' | 'maintenance' | 'reserved' | 'upcoming';

@Injectable()
export class UnitsService {
  constructor(private readonly tenantPrisma: TenantPrismaService) {}

  list(companyId: string, includeArchived = false, query?: PagedQueryDto) {
    // الحالةُ محسوبةٌ لكل وحدة (عقدٌ ساري · قادم · حجز) فتُقطَع القائمةُ في الذاكرة بعد الحساب.
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => pageArray(query, await this.listWithStatus(tx, includeArchived), UNIT_SEARCH_FIELDS));
  }

  create(companyId: string, dto: CreateUnitDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const property = await tx.property.findUnique({ where: { id: dto.propertyId } });
      if (!property) throw new NotFoundException(MSG.property_not_found);

      try {
        return await tx.unit.create({
          data: {
            propertyId: dto.propertyId,
            unitNo: dto.unitNo,
            floor: dto.floor,
            type: dto.type,
            subtype: dto.subtype,
            rentMonthly: dto.rentMonthly,
            roomDetails: dto.roomDetails,
            underMaintenance: dto.underMaintenance ?? false,
          } as unknown as Prisma.UnitCreateInput,
        });
      } catch (e) {
        if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === 'P2002') {
          throw new ConflictException(formatMessage(MSG.unit_number_taken, { unitNo: dto.unitNo }));
        }
        throw e;
      }
    });
  }

  /**
   * إنشاء بالجملة (§16): بادئة + عدد، كل شيء أو لا شيء — أي تصادم في
   * الأرقام يُرفض قبل إنشاء أي وحدة، ويُسمّي الأرقام المتصادمة.
   */
  createBulk(companyId: string, dto: BulkCreateUnitsDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const property = await tx.property.findUnique({ where: { id: dto.propertyId } });
      if (!property) throw new NotFoundException(MSG.property_not_found);

      const start = dto.startAt ?? 1;
      const numbers = Array.from({ length: dto.count }, (_, i) => `${dto.prefix}${start + i}`);
      // فحصان: الحيّ، ثم المُهمل في السلة — فهرس الفرادة يراهما معاً بينما
      // مرشّح الحذف الناعم يخفي الثاني عن الفحص العادي (اكتشاف المراجعة).
      const clashes = await tx.unit.findMany({ where: { propertyId: dto.propertyId, unitNo: { in: numbers } }, select: { unitNo: true } });
      const trashed = await tx.unit.findMany({ where: { propertyId: dto.propertyId, unitNo: { in: numbers }, deletedAt: { not: null } }, select: { unitNo: true } });
      if (clashes.length || trashed.length) {
        const names = [...clashes.map((c) => c.unitNo), ...trashed.map((c) => c.unitNo + ' (في سلة المحذوفات)')];
        throw new ConflictException(formatMessage(MSG.bulk_units_collision, { units: names.join('، ') }));
      }

      try {
      await tx.unit.createMany({
        data: numbers.map((unitNo) => ({
          companyId,
          propertyId: dto.propertyId,
          unitNo,
          floor: dto.floor,
          type: dto.type,
          subtype: dto.subtype,
          rentMonthly: dto.rentMonthly,
        })),
      });
      } catch (e) {
        // سباق مع إنشاء متزامن بعد الفحص — نفس الرد المسمّى لا 500 خام.
        if (e instanceof Prisma.PrismaClientKnownRequestError && e.code === 'P2002') {
          throw new ConflictException(formatMessage(MSG.bulk_units_collision, { units: dto.prefix + '…' }));
        }
        throw e;
      }
      return { created: numbers.length, unitNos: numbers };
    });
  }

  update(companyId: string, id: string, dto: UpdateUnitDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await tx.unit.findUnique({ where: { id } });
      if (!existing) throw new NotFoundException(MSG.unit_not_found);

      const data: Prisma.UnitUncheckedUpdateInput = {};
      if (dto.floor !== undefined) data.floor = dto.floor;
      if (dto.type !== undefined) data.type = dto.type;
      if (dto.subtype !== undefined) data.subtype = dto.subtype;
      if (dto.rentMonthly !== undefined) data.rentMonthly = dto.rentMonthly;
      if (dto.underMaintenance !== undefined) data.underMaintenance = dto.underMaintenance;
      if (dto.roomDetails !== undefined) data.roomDetails = dto.roomDetails;

      return tx.unit.update({ where: { id }, data });
    });
  }

  /** Hidden from day-to-day lists while staying in the books — archiving
   * is presentation, deletion is the trash. */
  setArchived(companyId: string, id: string, archived: boolean) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await tx.unit.findUnique({ where: { id } });
      if (!existing) throw new NotFoundException(MSG.unit_not_found);
      return tx.unit.update({ where: { id }, data: { archived } });
    });
  }

  async remove(companyId: string, id: string): Promise<void> {
    await this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await tx.unit.findUnique({ where: { id } });
      if (!existing) throw new NotFoundException(MSG.unit_not_found);
      // Checked explicitly rather than by catching a foreign-key error: a soft
      // delete never touches the FK, so the database can no longer be the one
      // refusing this. Same rule (CLAUDE.md 6.4), enforced where it's visible.
      // Counts historical contracts too — a unit that was ever let is history.
      const contractCount = await tx.contract.count({ where: { unitId: id } });
      if (contractCount > 0) throw new ConflictException(MSG.unit_has_contracts);

      // Soft delete — recoverable from the trash for 30 days.
      await tx.unit.update({ where: { id }, data: { deletedAt: new Date() } });
    });
  }

  /** Attaches a live-computed status to each unit — never stored, matching
   * the mockup's `unitStatusInfo`: under maintenance takes priority, then
   * whether a non-cancelled contract currently covers today, then whether an
   * active (non-expired) reservation holds the unit. */
  private async listWithStatus(tx: Prisma.TransactionClient, includeArchived = false): Promise<(Unit & { status: UnitStatus; currentTenant: string | null; upcomingTenant: string | null; reservedBy: string | null })[]> {
    await expireStaleReservations(tx);
    // `id` ختماً للترتيب التامّ (`ق-٣٥٦`) — القائمةُ تُقطَع في الذاكرة بعد اشتقاق الحالة.
    const units = await tx.unit.findMany({ ...(includeArchived ? {} : { where: { archived: false } }), orderBy: [{ propertyId: 'asc' }, { unitNo: 'asc' }, { id: 'asc' }] });
    const today = new Date();

    const activeContracts = await tx.contract.findMany({
      where: { start: { lte: today }, end: { gte: today }, status: { not: 'CANCELLED' } },
      select: { unitId: true, tenant: true },
    });
    const tenantByUnit = new Map(activeContracts.map((c) => [c.unitId, c.tenant]));

    // «محجوزة لعقد قادم مؤكَّد» (§16): عقدٌ لم يبدأ بعدُ وغير ملغى يحجز
    // الوحدة أقوى من حجز عادي — العقد سيحدث، والحجز قد لا يحدث.
    const upcomingContracts = await tx.contract.findMany({
      // DRAFT مستثنى عمداً: «المسودة قد لا تحدث، والقادم سيحدث» (تعليق المخطط
      // نفسه) — الشارة تقول «مؤكَّد» فلا يرفعها إلا عقدٌ مؤكَّد.
      where: { start: { gt: today }, status: { notIn: ['CANCELLED', 'DRAFT'] } },
      select: { unitId: true, tenant: true },
    });
    const upcomingByUnit = new Map(upcomingContracts.map((c) => [c.unitId, c.tenant]));

    const activeReservations = await tx.reservation.findMany({ where: { status: 'ACTIVE' }, select: { unitId: true, name: true } });
    const reservationByUnit = new Map(activeReservations.map((r) => [r.unitId, r.name]));

    return units.map((u) => {
      const tenant = tenantByUnit.get(u.id) ?? null;
      const upcomingTenant = upcomingByUnit.get(u.id) ?? null;
      const reservedBy = reservationByUnit.get(u.id) ?? null;
      const status: UnitStatus = u.underMaintenance ? 'maintenance' : tenant ? 'rented' : upcomingTenant ? 'upcoming' : reservedBy ? 'reserved' : 'vacant';
      return { ...u, status, currentTenant: tenant, upcomingTenant: status === 'upcoming' ? upcomingTenant : null, reservedBy: status === 'reserved' ? reservedBy : null };
    });
  }
}

import { IsDateString, IsIn, IsNumber, IsOptional, IsPositive, IsString, IsUUID, Matches } from 'class-validator';
import { CurrencyFieldsDto } from '../../common/currency/currency-fields.dto';

const PRIORITIES = ['URGENT', 'HIGH', 'NORMAL'] as const;

/** At least one non-whitespace character — `@IsNotEmpty` passes a run of
 * spaces, which is not a title. */
const NON_BLANK = /\S/;

export class CreateMaintenanceDto {
  @IsUUID()
  propertyId!: string;

  /// Optional: a fault in a shared area (lift, generator, roof) belongs to the
  /// property, not to any one unit.
  @IsOptional()
  @IsUUID()
  unitId?: string;

  @IsString()
  @Matches(NON_BLANK)
  title!: string;

  @IsOptional()
  @IsString()
  description?: string;

  @IsOptional()
  @IsIn(PRIORITIES)
  priority?: (typeof PRIORITIES)[number];

  @IsOptional()
  @IsString()
  reportedBy?: string;

  @IsDateString()
  reportedAt!: string;
}

/** Editable while the work is still open — see the status guard in the service. */
export class UpdateMaintenanceDto {
  @IsOptional()
  @IsString()
  @Matches(NON_BLANK)
  title?: string;

  @IsOptional()
  @IsString()
  description?: string;

  @IsOptional()
  @IsIn(PRIORITIES)
  priority?: (typeof PRIORITIES)[number];

  @IsOptional()
  @IsString()
  reportedBy?: string;

  @IsOptional()
  @IsString()
  notes?: string;
}

export class AssignMaintenanceDto {
  /// An employee or an outside contractor, so free text rather than an id.
  @IsString()
  @Matches(NON_BLANK)
  assignedTo!: string;
}

export class StartMaintenanceDto {
  @IsDateString()
  startedAt!: string;
}

export class CompleteMaintenanceDto extends CurrencyFieldsDto {
  @IsDateString()
  completedAt!: string;

  /// Omitted when the work cost the landlord nothing — an in-house fix, or one
  /// the contractor covered under warranty. Nothing is posted in that case.
  @IsOptional()
  @IsNumber()
  @IsPositive()
  cost?: number;

  @IsOptional()
  @IsString()
  notes?: string;
}

export class CloseMaintenanceDto {
  @IsDateString()
  closedAt!: string;
}

export class CancelMaintenanceDto {
  /// A cancelled request with no reason is indistinguishable from a forgotten
  /// one.
  @IsString()
  @Matches(NON_BLANK)
  cancelReason!: string;
}

import { Body, Controller, Get, Param, Patch, Post, Query, UseGuards } from '@nestjs/common';

import { PagedQueryDto } from '../common/dto/paged-query.dto';

import {
  AssignMaintenanceDto,
  CancelMaintenanceDto,
  CloseMaintenanceDto,
  CompleteMaintenanceDto,
  CreateMaintenanceDto,
  StartMaintenanceDto,

} from './dto/maintenance.dto';
import { ClaimWarrantyDto } from '../warranties/dto/warranty.dto';
import { MaintenanceService } from './maintenance.service';

/** Each step of the lifecycle is its own endpoint rather than a generic
 * status PATCH: they take different data (an assignee, a date, a cost), and a
 * free-form status field would let a caller jump the queue. */
@Controller('maintenance')
@UseGuards(JwtAuthGuard, TenantGuard)
@RequiresScreen('maintenance')
export class MaintenanceController {
  constructor(private readonly maintenanceService: MaintenanceService) {}

  @Get()
  @RequiresAction('view')
  list(@CurrentTenant() tenant: TenantContext, @Query() query: PagedQueryDto) {
    return this.maintenanceService.list(tenant.companyId, query);
  }

  @Get(':id')
  @RequiresAction('view')
  findOne(@CurrentTenant() tenant: TenantContext, @Param('id') id: string) {
    return this.maintenanceService.findOne(tenant.companyId, id);
  }

  @Post()
  @RequiresAction('add')
  create(@CurrentTenant() tenant: TenantContext, @Body() dto: CreateMaintenanceDto) {
    return this.maintenanceService.create(tenant.companyId, dto);
  }

  @Patch(':id')
  @RequiresAction('edit')
  update(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: UpdateMaintenanceDto) {
    return this.maintenanceService.update(tenant.companyId, id, dto);
  }

  @Post(':id/assign')
  assign(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: AssignMaintenanceDto) {
    return this.maintenanceService.assign(tenant.companyId, id, dto);
  }

  @Post(':id/start')
  start(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: StartMaintenanceDto) {
    return this.maintenanceService.start(tenant.companyId, id, dto);
  }

  /** Finishes the work and posts its cost. */
  @Post(':id/complete')
  @RequiresAction('record')
  complete(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: CompleteMaintenanceDto) {
    return this.maintenanceService.complete(tenant.companyId, id, dto);
  }

  @Post(':id/close')
  @RequiresAction('review')
  close(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: CloseMaintenanceDto) {
    return this.maintenanceService.close(tenant.companyId, id, dto);
  }

  @Post(':id/cancel')
  cancel(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: CancelMaintenanceDto) {
    return this.maintenanceService.cancel(tenant.companyId, id, dto);
  }

  /** Claims the job under a warranty — or clears the claim if cover was
   * refused. Send `warrantyId: null` to clear. */
  @Patch(':id/warranty')
  @RequiresAction('edit')
  claim(@CurrentTenant() tenant: TenantContext, @Param('id') id: string, @Body() dto: ClaimWarrantyDto) {
    return this.maintenanceService.claim(tenant.companyId, id, dto.warrantyId ?? null);
  }
}

import { BadRequestException, ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { MaintenanceStatus, Prisma } from '@prisma/client';
import { toBaseAmount } from '../common/currency/base-amount.util';
import { formatMessage, MSG } from '../common/i18n/messages';
import { PagedQueryDto } from '../common/dto/paged-query.dto';
import { pagedFindMany, searchTerm, searchWhere } from '../common/pagination/paged.util';
import { retryOnUniqueConflict } from '../common/retry-on-conflict.util';
import { TenantPrismaService } from '../common/tenant/tenant-prisma.service';
import { postMaintenanceCostEntry } from '../journal/auto-posting.util';
import { coverForTx } from '../warranties/warranties.service';
import {
  AssignMaintenanceDto,
  CancelMaintenanceDto,
  CloseMaintenanceDto,
  CompleteMaintenanceDto,
  CreateMaintenanceDto,
  StartMaintenanceDto,
  UpdateMaintenanceDto,
} from './dto/maintenance.dto';

const REQUEST_INCLUDE = {
  property: { select: { id: true, name: true } },
  unit: { select: { id: true, unitNo: true } },
};

/** حقولُ البحث المُعلَنة (`ق-١٢٣٦`) — العنوانُ والوصفُ ومَن بلّغ ومَن كُلِّف. */
const MAINTENANCE_SEARCH_FIELDS = ['title', 'description', 'reportedBy', 'assignedTo'] as const;

/**
 * What each status may become.
 *
 * Written as data rather than scattered `if`s so the whole lifecycle is legible
 * in one place — and so an illegal jump (reporting straight to CLOSED, skipping
 * the work) is impossible by construction rather than by remembering to check.
 */
const ALLOWED_TRANSITIONS: Record<MaintenanceStatus, MaintenanceStatus[]> = {
  REPORTED: ['ASSIGNED', 'IN_PROGRESS', 'CANCELLED'],
  ASSIGNED: ['IN_PROGRESS', 'CANCELLED'],
  IN_PROGRESS: ['DONE', 'CANCELLED'],
  // Reopening a finished job is a new request, not a rewind: the cost is
  // already posted and the history has to stay legible.
  DONE: ['CLOSED'],
  CLOSED: [],
  CANCELLED: [],
};

const STATUS_LABEL: Record<MaintenanceStatus, string> = {
  REPORTED: 'بلاغ جديد',
  ASSIGNED: 'مُسنَد',
  IN_PROGRESS: 'قيد التنفيذ',
  DONE: 'منجز',
  CLOSED: 'مغلق',
  CANCELLED: 'ملغى',
};

@Injectable()
export class MaintenanceService {
  constructor(private readonly tenantPrisma: TenantPrismaService) {}

  list(companyId: string, query?: PagedQueryDto) {
    const where: Prisma.MaintenanceRequestWhereInput = { ...searchWhere(searchTerm(query), MAINTENANCE_SEARCH_FIELDS) };
    return this.tenantPrisma.runInTenantContext(companyId, (tx) =>
      pagedFindMany(query, tx.maintenanceRequest, {
        where,
        include: REQUEST_INCLUDE,
        // Urgent first, then newest: the list is a work queue, not an archive.
        orderBy: [{ priority: 'asc' }, { reportedAt: 'desc' }],
      }),
    );
  }

  /**
   * One request, together with **what was under warranty when the fault was
   * reported**.
   *
   * The cover is computed against `reportedAt`, not today: a fault reported
   * while cover was live stays claimable even if it is opened here weeks
   * later. This is the whole reason warranties are recorded — a landlord who
   * finds out afterwards has already paid the contractor.
   */
  findOne(companyId: string, id: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const found = await tx.maintenanceRequest.findUnique({
        where: { id },
        include: { ...REQUEST_INCLUDE, warranty: { include: { supplier: { select: { id: true, name: true } } } } },
      });
      if (!found) throw new NotFoundException(MSG.maintenance_not_found);

      const availableCover = await coverForTx(tx, found.propertyId, found.unitId ?? undefined, found.reportedAt);
      return { ...found, availableCover };
    });
  }

  create(companyId: string, dto: CreateMaintenanceDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const property = await tx.property.findUnique({ where: { id: dto.propertyId } });
      if (!property) throw new NotFoundException(MSG.property_not_found);

      if (dto.unitId) {
        const unit = await tx.unit.findUnique({ where: { id: dto.unitId } });
        // A unit from a different property would put the request on a building
        // nobody is looking at.
        if (!unit || unit.propertyId !== dto.propertyId) throw new BadRequestException(MSG.unit_not_in_property);
      }

      const created = await tx.maintenanceRequest.create({
        data: {
          propertyId: dto.propertyId,
          unitId: dto.unitId ?? null,
          title: dto.title.trim(),
          description: dto.description?.trim() || null,
          priority: dto.priority ?? 'NORMAL',
          reportedBy: dto.reportedBy?.trim() || null,
          reportedAt: new Date(dto.reportedAt),
        } as unknown as Prisma.MaintenanceRequestUncheckedCreateInput,
      });

      return tx.maintenanceRequest.findUniqueOrThrow({ where: { id: created.id }, include: REQUEST_INCLUDE });
    });
  }

  /** Details stay editable until the job is finished — a tenant calling back
   * with more information is normal, and a closed job's record is not. */
  update(companyId: string, id: string, dto: UpdateMaintenanceDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await this.load(tx, id);
      if (existing.status === 'CLOSED' || existing.status === 'CANCELLED') {
        throw new ConflictException(formatMessage(MSG.maintenance_status_final, { status: STATUS_LABEL[existing.status] }));
      }

      return tx.maintenanceRequest.update({
        where: { id },
        data: {
          ...(dto.title !== undefined ? { title: dto.title.trim() } : {}),
          ...(dto.description !== undefined ? { description: dto.description.trim() || null } : {}),
          ...(dto.priority !== undefined ? { priority: dto.priority } : {}),
          ...(dto.reportedBy !== undefined ? { reportedBy: dto.reportedBy.trim() || null } : {}),
          ...(dto.notes !== undefined ? { notes: dto.notes.trim() || null } : {}),
        },
        include: REQUEST_INCLUDE,
      });
    });
  }

  assign(companyId: string, id: string, dto: AssignMaintenanceDto) {
    return this.transition(companyId, id, 'ASSIGNED', { assignedTo: dto.assignedTo.trim() });
  }

  start(companyId: string, id: string, dto: StartMaintenanceDto) {
    return this.transition(companyId, id, 'IN_PROGRESS', { startedAt: new Date(dto.startedAt) });
  }

  /**
   * Marks the work finished and posts what it cost.
   *
   * The cost lands here rather than at report time because that is the first
   * moment it is actually known — and a request that ends up cancelled should
   * never have touched the ledger at all.
   */
  complete(companyId: string, id: string, dto: CompleteMaintenanceDto) {
    return retryOnUniqueConflict(() =>
      this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
        const existing = await this.load(tx, id, { property: true });
        this.assertTransition(existing.status, 'DONE');

        const exchangeRate = dto.exchangeRate ?? 1;
        const baseCost = dto.cost != null ? toBaseAmount(dto.cost, exchangeRate) : null;

        const journalEntry =
          baseCost != null
            ? await postMaintenanceCostEntry(tx, companyId, {
                propertyId: existing.propertyId,
                unitId: existing.unitId,
                title: existing.title,
                propertyName: existing.property.name,
                date: new Date(dto.completedAt),
                baseAmount: baseCost,
              })
            : null;

        return tx.maintenanceRequest.update({
          where: { id },
          data: {
            status: 'DONE',
            completedAt: new Date(dto.completedAt),
            ...(dto.cost != null
              ? { cost: dto.cost, currency: dto.currency?.trim() || 'SAR', exchangeRate, baseCost, journalEntryId: journalEntry?.id ?? null }
              : {}),
            ...(dto.notes !== undefined ? { notes: dto.notes.trim() || null } : {}),
          },
          include: REQUEST_INCLUDE,
        });
      }),
    );
  }

  close(companyId: string, id: string, dto: CloseMaintenanceDto) {
    return this.transition(companyId, id, 'CLOSED', { closedAt: new Date(dto.closedAt) });
  }

  cancel(companyId: string, id: string, dto: CancelMaintenanceDto) {
    return this.transition(companyId, id, 'CANCELLED', { cancelReason: dto.cancelReason.trim() });
  }

  /**
   * Records that this job is being claimed under a warranty — or clears the
   * claim when the provider refuses cover.
   *
   * Kept separate from `update` because it is a different kind of fact: the
   * others describe the fault, this one describes who is expected to pay for
   * it. Allowed after completion too, since cover is often argued about long
   * after the work is done.
   */
  claim(companyId: string, id: string, warrantyId: string | null) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await this.load(tx, id);

      if (warrantyId) {
        const warranty = await tx.warranty.findUnique({ where: { id: warrantyId } });
        if (!warranty) throw new NotFoundException(MSG.warranty_not_found);

        // Checked rather than trusted: a claim recorded against cover that
        // never applied is worse than no record — it is a defence that
        // collapses the moment the provider looks at the dates.
        const covers =
          warranty.propertyId === existing.propertyId &&
          (warranty.unitId === null || warranty.unitId === existing.unitId) &&
          warranty.startDate <= existing.reportedAt &&
          warranty.endDate >= existing.reportedAt;
        if (!covers) throw new BadRequestException(MSG.warranty_not_covering);
      }

      return tx.maintenanceRequest.update({ where: { id }, data: { warrantyId }, include: REQUEST_INCLUDE });
    });
  }

  private transition(companyId: string, id: string, next: MaintenanceStatus, data: Prisma.MaintenanceRequestUncheckedUpdateInput) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await this.load(tx, id);
      this.assertTransition(existing.status, next);

      return tx.maintenanceRequest.update({ where: { id }, data: { status: next, ...data }, include: REQUEST_INCLUDE });
    });
  }

  /** Names both ends of the illegal move, so the user can see what they tried
   * to do rather than just being told "no". */
  private assertTransition(from: MaintenanceStatus, to: MaintenanceStatus): void {
    if (!ALLOWED_TRANSITIONS[from].includes(to)) {
      throw new ConflictException(formatMessage(MSG.maintenance_invalid_transition, { from: STATUS_LABEL[from], to: STATUS_LABEL[to] }));
    }
  }

  private async load(tx: Prisma.TransactionClient, id: string, include?: { property: true }) {
    const found = await tx.maintenanceRequest.findUnique({ where: { id }, include });
    if (!found) throw new NotFoundException(MSG.maintenance_not_found);
    return found as typeof found & { property: { name: string } };
  }
}

import { ConflictException, Injectable, NotFoundException } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { TenantPrismaService } from '../common/tenant/tenant-prisma.service';
import { CreateReservationDto } from './dto/create-reservation.dto';
import { formatMessage, MSG } from '../common/i18n/messages';
import { PagedQueryDto } from '../common/dto/paged-query.dto';
import { pagedFindMany, searchTerm, searchWhere } from '../common/pagination/paged.util';

const RESERVATION_INCLUDE = { unit: { include: { property: true } } };

/** حقولُ البحث المُعلَنة (`ق-١٢٣٦`) — اسمُ الحاجز وهاتفُه. */
const RESERVATION_SEARCH_FIELDS = ['name', 'phone'] as const;

@Injectable()
export class ReservationsService {
  constructor(private readonly tenantPrisma: TenantPrismaService) {}

  list(companyId: string, query?: PagedQueryDto) {
    const where: Prisma.ReservationWhereInput = { ...searchWhere(searchTerm(query), RESERVATION_SEARCH_FIELDS) };
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      await expireStaleReservations(tx);
      return pagedFindMany(query, tx.reservation, { where, include: RESERVATION_INCLUDE, orderBy: { createdAt: 'desc' } });
    });
  }

  create(companyId: string, dto: CreateReservationDto) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const unit = await tx.unit.findUnique({ where: { id: dto.unitId } });
      if (!unit) throw new NotFoundException(MSG.unit_not_found);

      const today = new Date();
      const currentContract = await tx.contract.findFirst({
        where: { unitId: dto.unitId, status: { not: 'CANCELLED' }, start: { lte: today }, end: { gte: today } },
      });
      if (currentContract) throw new ConflictException(MSG.reservation_unit_rented);

      await expireStaleReservations(tx);
      const existing = await activeReservationForUnit(tx, dto.unitId);
      if (existing) {
        throw new ConflictException(
          formatMessage(MSG.reservation_already_active, { expiryDate: existing.expiryDate.toISOString().slice(0, 10) }),
        );
      }

      const created = await tx.reservation.create({
        data: {
          unitId: dto.unitId,
          name: dto.name,
          phone: dto.phone,
          deposit: dto.deposit ?? 0,
          expiryDate: new Date(dto.expiryDate),
          status: 'ACTIVE',
        } as unknown as Prisma.ReservationUncheckedCreateInput,
      });

      return tx.reservation.findUniqueOrThrow({ where: { id: created.id }, include: RESERVATION_INCLUDE });
    });
  }

  cancel(companyId: string, id: string) {
    return this.tenantPrisma.runInTenantContext(companyId, async (tx) => {
      const existing = await tx.reservation.findUnique({ where: { id } });
      if (!existing) throw new NotFoundException(MSG.reservation_not_found);
      return tx.reservation.update({ where: { id }, data: { status: 'EXPIRED' }, include: RESERVATION_INCLUDE });
    });
  }
}

/** Matches the mockup's `expireOldReservations` — flips any ACTIVE reservation
 * whose expiry date has passed, called lazily on every read/write rather than
 * via a background job (same pattern as ContractsService.refreshInstallmentStatuses). */
export async function expireStaleReservations(tx: Prisma.TransactionClient): Promise<void> {
  await tx.reservation.updateMany({
    where: { status: 'ACTIVE', expiryDate: { lt: new Date() } },
    data: { status: 'EXPIRED' },
  });
}

/** Matches the mockup's `unitActiveReservation` (post-expiry). Exported so
 * ContractsService can check/convert reservations when creating a contract. */
export async function activeReservationForUnit(tx: Prisma.TransactionClient, unitId: string) {
  return tx.reservation.findFirst({ where: { unitId, status: 'ACTIVE' } });
}
